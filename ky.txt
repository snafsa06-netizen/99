local _ActiveTweensByInstance = setmetatable({}, {__mode = "k"})

local function SafeTween(instance, tweenInfo, properties)
    local existing = _ActiveTweensByInstance[instance]
    if existing then
        for propName, t in pairs(existing) do
            if properties[propName] ~= nil and t then
                t:Cancel()
                existing[propName] = nil
            end
        end
    else
        existing = {}
        _ActiveTweensByInstance[instance] = existing
    end

    local tween = TweenService:Create(instance, tweenInfo, properties)
    for propName in pairs(properties) do
        existing[propName] = tween
    end
    tween:Play()
    return tween
end

local MuteManager = {
    OriginalVolumes = {},
    ActiveTweens = {},
    IsMuting = false,
    FadeDuration = 1,
    Connection = nil,
}

local function IsUiSound(sound)
    if not sound or not sound.Parent then return true end
    if sound:IsDescendantOf(CoreGui) then return true end
  
    if sound.Name == "IntroMusic" or sound.Name == "activeClickSound" or sound.Name == "activePageSound" then
      return true
    end
    return false
end

local function StoreOriginalVolume(sound)
    if not MuteManager.OriginalVolumes[sound] then
        MuteManager.OriginalVolumes[sound] = sound.Volume
    end
end

local function TweenSoundVolume(sound, targetVolume, duration, callback)
    if not sound or not sound.Parent then 
        if callback then callback() end
        return 
    end
    
    if MuteManager.ActiveTweens[sound] then
        MuteManager.ActiveTweens[sound]:Cancel()
        MuteManager.ActiveTweens[sound] = nil
    end
    
    local tween = TweenService:Create(sound, TweenInfo.new(duration, Enum.EasingStyle.Linear), {Volume = targetVolume})
    MuteManager.ActiveTweens[sound] = tween
    tween:Play()
    if callback then
        tween.Completed:Connect(callback)
    end
end

local function ProcessAllSounds(targetVolume, duration)
    for sound, origVol in pairs(MuteManager.OriginalVolumes) do
        if sound and sound.Parent and not IsUiSound(sound) then
            local vol = (targetVolume == 0) and 0 or origVol
            TweenSoundVolume(sound, vol, duration)
        else
            MuteManager.OriginalVolumes[sound] = nil
        end
    end
end

local function OnDescendantAdded(obj)
    pcall(function()
        if (obj:IsA("Sound") or obj:IsA("SoundGroup")) and not IsUiSound(obj) then
            StoreOriginalVolume(obj)
            if MuteManager.IsMuting then
                obj.Volume = 0
            end
        end
    end)
end

function MuteManager:Init()
    for _, obj in pairs(game:GetDescendants()) do
        if (obj:IsA("Sound") or obj:IsA("SoundGroup")) and not IsUiSound(obj) then
            StoreOriginalVolume(obj)
            if self.IsMuting then
                obj.Volume = 0
            end
        end
    end
    
    if self.Connection then self.Connection:Disconnect() end
    self.Connection = game.DescendantAdded:Connect(OnDescendantAdded)
end

function MuteManager:FadeOutSounds(duration)
    self.FadeDuration = duration or 1
    self.IsMuting = true
    
    if not self.Connection then
        self:Init()
    end
    
    ProcessAllSounds(0, self.FadeDuration)
end

function MuteManager:FadeInSounds(duration)
    self.FadeDuration = duration or 1
    self.IsMuting = false
    
    for sound, origVol in pairs(self.OriginalVolumes) do
        if sound and sound.Parent and not IsUiSound(sound) then
            TweenSoundVolume(sound, origVol, self.FadeDuration)
        else
            self.OriginalVolumes[sound] = nil
        end
    end
end

function MuteManager:Cleanup()
    if self.Connection then
        self.Connection:Disconnect()
        self.Connection = nil
    end
    for sound, tween in pairs(self.ActiveTweens) do
        if tween then tween:Cancel() end
    end
    self.ActiveTweens = {}
    self.OriginalVolumes = {}
    self.IsMuting = false
end

-- دالة تحديث المربع اليميني ومعالجة الشفافية والتباعد للصور الأخرى
local function SelectGame(imgBtn, name, url, assetId, detailId)
    local ti = TweenInfo.new(0.25, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    local tiBack = TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out)

    if SelectedGameCard == imgBtn then return end

    SelectedGameCard = imgBtn
    SelectedGameUrl = url
    SelectedGameName = name

    -- حساب موضع الصورة المضغوط عليها لدفع الصور الأخرى بعيداً عنها
    local selectedPos = imgBtn.Parent.AbsolutePosition
    for _, otherImg in ipairs(GameImagesList) do
        local otherScale = otherImg:FindFirstChildOfClass("UIScale")
        if otherImg ~= imgBtn then
            if otherScale then
                SafeTween(otherScale, ti, {Scale = 1.0})
            end
            local otherPos = otherImg.Parent.AbsolutePosition
            local dir = (otherPos - selectedPos).Unit
            if dir.Magnitude == 0 or dir.X ~= dir.X then
                dir = Vector2.new(0, 1)
            end
            -- إزاحة خفيفة بمقدار 8 بكسل والشفافية تصبح 0.60
            local targetPos = UDim2.new(0, dir.X * 8, 0, dir.Y * 8)
            SafeTween(otherImg, ti, {
                ImageTransparency = 0.60,
                Position = targetPos
            })
        else
            -- الصورة المحددة تصبح واضحة تماماً وتكبر لـ 1.13 وترجع لمركزها الطبيعي
            if otherScale then
                SafeTween(otherScale, tiBack, {Scale = 1.13})
            end
            SafeTween(otherImg, ti, {
                ImageTransparency = 0,
                Position = UDim2.new(0, 0, 0, 0)
            })
        end
    end

    local tiFast = TweenInfo.new(0.15, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    local tiIn   = TweenInfo.new(0.3,  Enum.EasingStyle.Sine, Enum.EasingDirection.Out)

    SafeTween(DetailImage, tiFast, {ImageTransparency = 1})
    SafeTween(DetailNameLabel, tiFast, {TextTransparency  = 1})
    SafeTween(AntiBanLabel, tiFast, {TextTransparency  = 1}) -- إخفاء مؤقتاً

    LaunchBtn.BackgroundColor3 = TargetGlowColor
    SafeTween(LaunchBtn, tiIn, {BackgroundTransparency = 0.9, TextTransparency = 0})

    task.delay(0.15, function()
        DetailImage.Image    = "rbxthumb://type=Asset&id=" .. detailId .. "&w=420&h=420"
        DetailNameLabel.Text = name

        if name == "Combat Warriors" then
            AntiBanLabel.Text       = "Under Development"
            AntiBanLabel.TextColor3 = Color3.fromRGB(255, 220, 0)
        else
            AntiBanLabel.Text       = "Anti-ban"
            AntiBanLabel.TextColor3 = Color3.fromRGB(0, 255, 0)
        end

        local tiIn2 = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
        SafeTween(DetailImage, tiIn2, {ImageTransparency = 0})
        SafeTween(DetailNameLabel, tiIn2, {TextTransparency  = 0})
        SafeTween(AntiBanLabel, tiIn2, {TextTransparency  = 0})
    end)
end

local function AddSmoothAnim(btn, clickCallback, isColorSquare)
    local scale = btn:FindFirstChildOfClass("UIScale") or Instance.new("UIScale", btn)
    
    local isSizeBtn = (btn.Name == "SizeSmallBtn" or btn.Name == "SizeLargeBtn")
    -- الشفافية "الطبيعية" لأزرار الحجم هي 0.85 (كما تضعها RefreshSizeButtonsVisuals)
    local SIZE_BTN_NORMAL_TRANS = 0.85
    local SIZE_BTN_HOVER_TRANS  = 0.40  -- أوضح عند hover (شفافية أقل)
    local SIZE_BTN_PRESS_TRANS  = 0.15  -- أوضح جداً عند الضغط
    local SIZE_BTN_NORMAL_SCALE = 1.0
    local SIZE_BTN_HOVER_SCALE  = 1.20  -- يكبر كثير عند hover
    local SIZE_BTN_PRESS_SCALE  = 1.10  -- يكبر أقل عند الضغط

    btn.MouseEnter:Connect(function()
        if (btn.Name == "SizeSmallBtn" and SizeIndex <= 7) or (btn.Name == "SizeLargeBtn" and SizeIndex == MaxSizeIndex) then return end
        
        if isSizeBtn then
            SafeTween(scale, TweenInfo.new(0.15, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = SIZE_BTN_HOVER_SCALE})
            SafeTween(btn, TweenInfo.new(0.15), {BackgroundTransparency = SIZE_BTN_HOVER_TRANS})
            return
        end

        local targetScale = (btn:GetAttribute("TargetScale") or 1.0) + 0.02
        SafeTween(scale, TweenInfo.new(0.12, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {Scale = targetScale})
        
        local hoverTrans = math.min(1, (btn:GetAttribute("TargetTrans") or 0.92) + 0.05)

        if btn.ClassName == "TextButton" and btn.BackgroundTransparency > 0 then
            SafeTween(btn, TweenInfo.new(0.12), {BackgroundTransparency = hoverTrans})
        end
        if isColorSquare and btn.Name ~= "ActiveColorBtn" then
            local stroke = btn:FindFirstChild("UIStroke")
            if stroke then SafeTween(stroke, TweenInfo.new(0.12), {Transparency = 0.6}) end
        end
    end)
    
    btn.MouseLeave:Connect(function()
        if isSizeBtn then
            local leaveTrans = ((btn.Name == "SizeSmallBtn" and SizeIndex <= 7) or (btn.Name == "SizeLargeBtn" and SizeIndex == MaxSizeIndex))
                and 0.95 or SIZE_BTN_NORMAL_TRANS
            SafeTween(scale, TweenInfo.new(0.18, Enum.EasingStyle.Sine), {Scale = SIZE_BTN_NORMAL_SCALE})
            SafeTween(btn, TweenInfo.new(0.18), {BackgroundTransparency = leaveTrans})
            return
        end

        local leaveScale = btn:GetAttribute("TargetScale") or 1.0
        SafeTween(scale, TweenInfo.new(0.12, Enum.EasingStyle.Sine), {Scale = leaveScale})
        
        local leaveTrans = btn:GetAttribute("TargetTrans") or 0.92

        if btn.ClassName == "TextButton" and btn.BackgroundTransparency > 0 then
            SafeTween(btn, TweenInfo.new(0.12), {BackgroundTransparency = leaveTrans})
        end
       
        if isColorSquare and btn.Name ~= "ActiveColorBtn" then
             local stroke = btn:FindFirstChild("UIStroke")
             if stroke then SafeTween(stroke, TweenInfo.new(0.12), {Transparency = 1}) end
        end
    end)
    
    btn.MouseButton1Down:Connect(function()
        if (btn.Name == "SizeSmallBtn" and SizeIndex <= 7) or (btn.Name == "SizeLargeBtn" and SizeIndex == MaxSizeIndex) then return end

        if isSizeBtn then
            -- عند الضغط: يكبر + يصير أوضح (عكس باقي الأزرار)
            SafeTween(scale, TweenInfo.new(0.08, Enum.EasingStyle.Sine), {Scale = SIZE_BTN_PRESS_SCALE})
            SafeTween(btn, TweenInfo.new(0.08), {BackgroundTransparency = SIZE_BTN_PRESS_TRANS})
            return
        end

        local pressScale = (btn:GetAttribute("TargetScale") or 1.0) - 0.04
        SafeTween(scale, TweenInfo.new(0.1, Enum.EasingStyle.Sine), {Scale = pressScale})
        local pressTrans = math.min(1, (btn:GetAttribute("TargetTrans") or 0.92) + 0.06)
        if btn.ClassName == "TextButton" and btn.BackgroundTransparency > 0 then
            SafeTween(btn, TweenInfo.new(0.08), {BackgroundTransparency = pressTrans})
        end
    end)
  
    btn.MouseButton1Click:Connect(function()
         if (btn.Name == "SizeSmallBtn" and SizeIndex <= 7) or (btn.Name == "SizeLargeBtn" and SizeIndex == MaxSizeIndex) then return end
        PlayClickSound()
        if clickCallback then clickCallback() end
    end)
end

local function SwitchThemeSubPage(targetPage, activeBtn)
    if CurrentThemeSubPage == targetPage and CurrentThemeSubPage ~= nil then return end
    local oldPage = CurrentThemeSubPage
    CurrentThemeSubPage = targetPage
 
    local ti = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    
    if oldPage then
        for _, el in ipairs(oldPage:GetDescendants()) do
            if el:IsA("TextLabel") or el:IsA("TextButton") then
                SafeTween(el, ti, {TextTransparency = 1, BackgroundTransparency = 1})
            elseif el:IsA("UIStroke") then
                SafeTween(el, ti, {Transparency = 1})
            end
        end
        task.delay(0.3, function() oldPage.Visible = false end)
    end
    
    if targetPage then
        targetPage.Visible = true
        for _, el in ipairs(targetPage:GetDescendants()) do
            if el:IsA("TextLabel") then
                SafeTween(el, ti, {TextTransparency = 0})
            elseif el:IsA("TextButton") then
                 SafeTween(el, ti, {TextTransparency = 0})
                 if el.Name == "SizeSmallBtn" or el.Name == "SizeLargeBtn" then

                elseif el.Name == "ActiveColorBtn" or el == NormalRedBtn or el == DarkRedBtn or el == BlueBtn or el == LightBlueBtn or el == PinkBtn or el == DarkPinkBtn or el == LightPinkBtn then
                     SafeTween(el, ti, {BackgroundTransparency = el:GetAttribute("TargetTrans") or 0})
                else
                    SafeTween(el, ti, {BackgroundTransparency = el:GetAttribute("TargetTrans") or 0.92})
                end
            elseif el:IsA("UIStroke") then
                if el.Parent and el.Parent.Name == "ActiveColorBtn" then
                     SafeTween(el, ti, {Transparency = 0})
                end
            end
        end
    
        if targetPage == SizeSubMenu then
            RefreshSizeButtonsVisuals()
        end
    end
end

local function SetupThemeTabInteractions(btn, targetSubMenu)
    local scaleObj = btn:FindFirstChildOfClass("UIScale")
    
    btn.MouseEnter:Connect(function()
        if ActiveThemeTab ~= btn then
            SafeTween(btn, TweenInfo.new(0.12), {TextTransparency = 0.15})
            if scaleObj then SafeTween(scaleObj, TweenInfo.new(0.12), {Scale = 1.0}) end
        end
    end)
    
    btn.MouseLeave:Connect(function()
        if ActiveThemeTab ~= btn then
            SafeTween(btn, TweenInfo.new(0.12), {TextTransparency = 0.5})
            local leaveScale = btn:GetAttribute("TargetScale") or 0.94
            if scaleObj then SafeTween(scaleObj, TweenInfo.new(0.12), {Scale = leaveScale}) end
        end
    end)
    
    btn.MouseButton1Click:Connect(function()
        PlayClickSound()
        ActiveThemeTab = btn
        UpdateThemeTabsVisuals()
        SwitchThemeSubPage(targetSubMenu, btn)
    end)
end

local function ApplyColorTheme(glowColor, activeBtn)
    TargetGlowColor = glowColor
    
    local r, g, b = glowColor.R, glowColor.G, glowColor.B
    local mainColor = Color3.new(r * 0.14, g * 0.14, b * 0.14)
    local boxColor = Color3.new(r * 0.08, g * 0.08, b * 0.08)
    
    CurrentMainBgColor = mainColor
    CurrentBoxBgColor = boxColor

    local ti = TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    
    SafeTween(MainFrame, ti, {BackgroundColor3 = mainColor})
    
    for _, box in pairs(AllUIBoxes) do
        if box and (box:IsA("TextButton") or box:IsA("TextBox") or box:IsA("Frame")) then 
            SafeTween(box, ti, {BackgroundColor3 = boxColor}) 
        end
    end

    for _, btn in pairs(SettingButtonsList) do
        if btn then SafeTween(btn, ti, {BackgroundColor3 = boxColor}) end
    end
    
    SafeTween(SizeSmallBtn, ti, {BackgroundColor3 = glowColor})
    SafeTween(SizeLargeBtn, ti, {BackgroundColor3 = glowColor})
    SafeTween(SettingsScroll, ti, {ScrollBarImageColor3 = glowColor})
    SafeTween(LangVerticalScroll, ti, {ScrollBarImageColor3 = glowColor})
    SafeTween(UpdatesScroll, ti, {ScrollBarImageColor3 = glowColor})
    SafeTween(GameScrollFrame, ti, {ScrollBarImageColor3 = glowColor})
    LaunchBtn.BackgroundColor3 = glowColor

    -- تحديث لون زر RUN وزر Back في DinoLaunchPanel عند تغيير الثيم
    if DinoRunBtn then
        DinoRunBtn.BackgroundColor3 = glowColor
    end
    if DinoCloseBtn then
        DinoCloseBtn.BackgroundColor3 = glowColor
    end
    if DinoTextScroll then
        SafeTween(DinoTextScroll, ti, {ScrollBarImageColor3 = glowColor})
    end
    
    UpdateLanguageVisuals() 
    UpdateThemeTabsVisuals()
    
    ActiveColorBtn.Name = "ColorBtn"
    ActiveColorBtn = activeBtn
    ActiveColorBtn.Name = "ActiveColorBtn"

    -- ====================================================================
    -- [إصلاح]: أزرار الألوان تتحرك بعيداً عن الزر المحدد بدل ما تغطي عليه
    -- كل زر له موضعه الأصلي (posX) محفوظ كـ Attribute، ونحسب إزاحة بسيطة
    -- بناءً على اتجاهه (يسار/يمين) من الزر النشط وحجم تكبيره (1.15 scale).
    -- ====================================================================
    local colorBtnsList = {NormalRedBtn, DarkRedBtn, BlueBtn, LightBlueBtn, PinkBtn, DarkPinkBtn, LightPinkBtn}
    local tiScale = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    local tiPos   = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)

    -- نحفظ الـ OriginalX لكل زر أول مرة (إن لم يكن محفوظاً)
    for _, btn in ipairs(colorBtnsList) do
        if not btn:GetAttribute("OriginalPosX") then
            btn:SetAttribute("OriginalPosX", btn.Position.X.Scale)
        end
    end

    -- نوجد index الزر النشط لمعرفة اتجاه الإزاحة
    local activeIdx = 1
    for i, btn in ipairs(colorBtnsList) do
        if btn == ActiveColorBtn then activeIdx = i; break end
    end

    -- مقدار الإزاحة الأفقية للأزرار المجاورة (كـ Scale)
    local PUSH_AMOUNT = 0.06

    for i, btn in ipairs(colorBtnsList) do
        local stroke = btn:FindFirstChildOfClass("UIStroke")
        local scaleObj = btn:FindFirstChildOfClass("UIScale") or Instance.new("UIScale", btn)
        local origX = btn:GetAttribute("OriginalPosX") or btn.Position.X.Scale
        local origY = btn.Position.Y.Scale

        if btn == ActiveColorBtn then
            btn:SetAttribute("TargetScale", 1.15)
            btn:SetAttribute("TargetTrans", 0.8)
            SafeTween(scaleObj, tiScale, {Scale = 1.15})
            SafeTween(btn, tiScale, {BackgroundTransparency = 0.8})
            -- الزر النشط يرجع لموضعه الأصلي
            SafeTween(btn, tiPos, {Position = UDim2.new(origX, 0, origY, 0)})
            if stroke then SafeTween(stroke, tiScale, {Transparency = 0}) end
        else
            btn:SetAttribute("TargetScale", 0.85)
            btn:SetAttribute("TargetTrans", 0)
            SafeTween(scaleObj, tiScale, {Scale = 0.85})
            SafeTween(btn, tiScale, {BackgroundTransparency = 0})
            if stroke then SafeTween(stroke, tiScale, {Transparency = 1}) end

            -- نحرك الزر بعيداً عن الزر النشط (يسار أو يمين)
            local pushDir = (i < activeIdx) and -1 or 1
            -- فقط الجيران المباشرين يتأثرون بالإزاحة
            local dist = math.abs(i - activeIdx)
            local pushX = origX + pushDir * PUSH_AMOUNT * (1 / dist)
            SafeTween(btn, tiPos, {Position = UDim2.new(pushX, 0, origY, 0)})
        end
    end
end

local function FormatTime(seconds)
    local h = math.floor(seconds / 3600)
    local m = math.floor((seconds % 3600) / 60)
    local s = seconds % 60
    return string.format("%02d:%02d:%02d", h, m, s)
end

local function UpdateNavigationFocus()
    local tiSmooth = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    
    TweenService:Create(ExitBtn:FindFirstChildOfClass("UIScale"), tiSmooth, {Scale = 0.94}):Play()
    SafeTween(ExitBtn, tiSmooth, {TextTransparency = 0.5})

    local buttons = {
        {btn = NavKeyBtn, page = "Key"},
        {btn = NavGamesBtn, page = "Games"},
        {btn = NavSettingsBtn, page = "Settings"}
    }

    for _, item in pairs(buttons) do
        local scaleObj = item.btn:FindFirstChildOfClass("UIScale") or Instance.new("UIScale", item.btn)
        if CurrentPage == item.page then

            item.btn:SetAttribute("TextTargetTrans", 0)
            SafeTween(scaleObj, tiSmooth, {Scale = 1.08})

         else

            item.btn:SetAttribute("TargetTrans", 1)
            item.btn:SetAttribute("TextTargetTrans", 0.5)
            SafeTween(scaleObj, tiSmooth, {Scale = 0.94})
            SafeTween(item.btn, tiSmooth, {TextTransparency = 0.5})
        end
    end

    if CurrentPage == "Key" and IsLoggedIn and RequireKeyForEU then
        TimerLabel.Visible = true
        SafeTween(TimerLabel, tiSmooth, {TextTransparency = 0})

        if TimerLabel.Visible then
            SafeTween(TimerLabel, tiSmooth, {TextTransparency = 1})
            task.delay(0.3, function() if CurrentPage ~= "Key" then TimerLabel.Visible = false end end)
        end
    end
end

-- نظام تفاعل الانيميشن المحدث للأزرار والصور لتطابق المؤثرات
-- [إصلاح]: flag يمنع MouseEnter/MouseLeave/MouseButton1Down من التدخل
-- بعد ضغط LAUNCH، لأنها تشغّل tweens تلغي توين الإخفاء في OpenDinoPanel
local _DinoOpening = false

local function AnimateButton(btn, callback)
    local uiScale = btn:FindFirstChildOfClass("UIScale") or Instance.new("UIScale", btn)

    local tweenInfoSlow = TweenInfo.new(0.22, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    local tweenInfoPress = TweenInfo.new(0.1, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)

    local isNavButton = (btn == NavKeyBtn or btn == NavGamesBtn or btn == NavSettingsBtn or btn == ExitBtn)
    local isLaunchBtn = (btn == LaunchBtn)
    local isGameButton = (not isLaunchBtn) and btn:IsDescendantOf(GameSelectionFrame)
    local isRightControl = (btn == ApplyLangBtn or btn == ResetLangBtn)
    local isKeySystemButton = (btn == SubmitBtn or btn == GetLinkBtn or btn == KeyInput)
    local isSettingBtn = btn:IsDescendantOf(SettingsScroll)

    local activeScaleTween = nil
    local activeBtnTween = nil

    local function playTween(targetScale, targetBgTrans, targetTextTrans, speedInfo)
         if activeScaleTween then activeScaleTween:Cancel() end

        activeScaleTween = TweenService:Create(uiScale, speedInfo, {Scale = targetScale})
        activeScaleTween:Play()

        -- [إصلاح] استخدام SafeTween يلغي تلقائياً أي توين سابق (حتى لو أُنشئ
        -- من دالة أخرى مثل OpenDinoPanel/CloseDinoPanel) على نفس الخصائص لهذا الزر
        activeBtnTween = SafeTween(btn, speedInfo, {
            BackgroundTransparency = targetBgTrans,
            TextTransparency = targetTextTrans
        })
     end

    btn.MouseEnter:Connect(function()
        if isGameButton and not IsLoggedIn then return end
        if isKeySystemButton and IsLoggedIn then return end 
    
        local baseScale = btn:GetAttribute("TargetScale") or 1.0
        local baseBg = btn:GetAttribute("TargetTrans") or 0.92
        local baseText = btn:GetAttribute("TextTargetTrans") or 0

        if isRightControl then 
             playTween(baseScale + 0.03, math.min(1, baseBg + 0.05), math.min(1, baseText + 0.1), tweenInfoSlow)
            return 
        end

        if isLaunchBtn then
            if _DinoOpening then return end
            if btn.BackgroundTransparency < 1 then
                playTween(baseScale + 0.04, 0.8, 0, tweenInfoSlow)
            end
            return
        end

        if isNavButton then
            playTween(baseScale + 0.05, 1, math.max(0, baseText - 0.2), tweenInfoSlow)
        elseif isSettingBtn then
            if activeScaleTween then activeScaleTween:Cancel() end
            activeScaleTween = TweenService:Create(uiScale, tweenInfoSlow, {Scale = baseScale + 0.02})
            activeScaleTween:Play()
            if btn.TextTransparency > 0 then
                if activeBtnTween then activeBtnTween:Cancel() end
                activeBtnTween = TweenService:Create(btn, tweenInfoSlow, {TextTransparency = math.max(0, baseText - 0.2)})
                activeBtnTween:Play()
            end
        elseif isGameButton then
            -- تأثير التصويب (Hover) الموحد للصور: تكبير وإضاءة
            local actualBase = (SelectedGameCard == btn) and 1.13 or 1.0
            if activeScaleTween then activeScaleTween:Cancel() end
            activeScaleTween = TweenService:Create(uiScale, tweenInfoSlow, {Scale = actualBase + 0.04})
            activeScaleTween:Play()
            
            if activeBtnTween then activeBtnTween:Cancel() end
            activeBtnTween = TweenService:Create(btn, tweenInfoSlow, {ImageTransparency = math.max(0, btn.ImageTransparency - 0.15)})
            activeBtnTween:Play()
        else
            if btn.Visible and btn.TextTransparency < 1 then
                 playTween(baseScale + 0.04, math.min(1, baseBg + 0.05), math.min(1, baseText + 0.1), tweenInfoSlow)
            end
        end
    end)

    btn.MouseLeave:Connect(function()
        if isGameButton and not IsLoggedIn then return end
        if isKeySystemButton and IsLoggedIn then return end
  
        local defScale = btn:GetAttribute("TargetScale") or 1.0
        local defBg = btn:GetAttribute("TargetTrans") or 0.92
        local defText = btn:GetAttribute("TextTargetTrans") or 0

        if isRightControl then 
            playTween(defScale, defBg, defText, tweenInfoSlow)
            return 
        end

        if isLaunchBtn then
            if _DinoOpening then return end
            -- [إصلاح]: لا نُعيد الزر ظاهراً إذا كان توين الإخفاء قد بدأ فعلاً
            -- (قد يكون BackgroundTransparency لم يصل 1 بعد لكن الزر في طور الاختفاء)
            if btn.BackgroundTransparency >= 1 then return end
            playTween(defScale, 0.9, 0, tweenInfoSlow)
            return
        end

        if isNavButton then
            playTween(defScale, 1, defText, tweenInfoSlow)
        elseif isSettingBtn then
            if activeScaleTween then activeScaleTween:Cancel() end
            activeScaleTween = TweenService:Create(uiScale, tweenInfoSlow, {Scale = defScale})
            activeScaleTween:Play()
            if activeBtnTween then activeBtnTween:Cancel() end
            activeBtnTween = TweenService:Create(btn, tweenInfoSlow, {TextTransparency = defText})
            activeBtnTween:Play()
        elseif isGameButton then
            -- إرجاع الصورة للحجم الطبيعي والشفافية الصحيحة المحددة
            local actualDef = (SelectedGameCard == btn) and 1.13 or 1.0
            local targetTrans = 0
            if not IsLoggedIn then
                targetTrans = 0.5
            elseif SelectedGameCard then
                targetTrans = (SelectedGameCard == btn) and 0 or 0.60
            end
            
            if activeScaleTween then activeScaleTween:Cancel() end
            activeScaleTween = TweenService:Create(uiScale, tweenInfoSlow, {Scale = actualDef})
            activeScaleTween:Play()
            
            if activeBtnTween then activeBtnTween:Cancel() end
            activeBtnTween = TweenService:Create(btn, tweenInfoSlow, {ImageTransparency = targetTrans})
            activeBtnTween:Play()
        else
            if IsLoggedIn and isKeySystemButton then
                playTween(defScale, 0.98, 0.7, tweenInfoSlow)
            else
                playTween(defScale, defBg, defText, tweenInfoSlow)
            end
        end
    end)

    btn.MouseButton1Down:Connect(function()
        if isGameButton and not IsLoggedIn then return end
        if isKeySystemButton and IsLoggedIn then return end
        
        local pressScale = (btn:GetAttribute("TargetScale") or 1.0) - 0.06
        
        if isSettingBtn then
            if activeScaleTween then activeScaleTween:Cancel() end
            activeScaleTween = TweenService:Create(uiScale, tweenInfoPress, {Scale = pressScale})
            activeScaleTween:Play()
        elseif isGameButton then
            -- تأثير الضغط (Press) الموحد للصور: انكماش خفيف للداخل لتبدو حية
            local actualBase = (SelectedGameCard == btn) and 1.13 or 1.0
            if activeScaleTween then activeScaleTween:Cancel() end
            activeScaleTween = TweenService:Create(uiScale, tweenInfoPress, {Scale = actualBase - 0.06})
            activeScaleTween:Play()
        elseif isLaunchBtn then
            if _DinoOpening then return end
            if btn.BackgroundTransparency < 1 then
                playTween(pressScale, 0.8, 0, tweenInfoPress)
            end
        elseif btn.Visible and btn.TextTransparency < 1 then
            local bgTrans = isNavButton and 1 or math.min(1, (btn:GetAttribute("TargetTrans") or 0.92) + 0.06)
            local targetTrans = isNavButton and 0.1 or math.min(1, (btn:GetAttribute("TextTargetTrans") or 0) + 0.15)
            playTween(pressScale, bgTrans, targetTrans, tweenInfoPress)
        end
    end)

    btn.MouseButton1Click:Connect(function()
        if isGameButton and not IsLoggedIn then return end
        if isKeySystemButton and IsLoggedIn then return end
        
        if isGameButton then
            local imgScaleObj = btn:FindFirstChildOfClass("UIScale")
            if imgScaleObj then
                 local returnScale = (SelectedGameCard == btn) and 1.13 or 1.0
                if activeScaleTween then activeScaleTween:Cancel() end
                activeScaleTween = TweenService:Create(imgScaleObj, tweenInfoSlow, {Scale = returnScale})
                activeScaleTween:Play()
            end
            PlayClickSound()
            if callback then callback() end
            return
        end

        if isLaunchBtn then
            if _DinoOpening then return end
            if btn.BackgroundTransparency < 1 then
                -- يرجع يكبر لحجم الهوفر (zoom in) بأنيميشن Back ثم يفتح البانيل
                local hoverScale = (btn:GetAttribute("TargetScale") or 1.0) + 0.04
                if activeScaleTween then activeScaleTween:Cancel() end
                activeScaleTween = TweenService:Create(uiScale,
                    TweenInfo.new(0.18, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
                    {Scale = hoverScale}
                )
                activeScaleTween:Play()
                PlayClickSound()
                task.delay(0.18, function()
                    if callback then callback() end
                end)
            end
            return
        end
        
        if btn.Visible and btn.TextTransparency < 1 and callback then
             PlayClickSound()
             callback()
        end
    end)
end

local EndTime = 0

local function StartTimer(endTime)
    if not RequireKeyForEU then return end 

    if CurrentPage == "Key" then
        TimerLabel.Visible = true
        SafeTween(TimerLabel, TweenInfo.new(0.5), {TextTransparency = 0})
    end
    task.spawn(function()
        while task.wait(1) do
             local remaining = endTime - os.time()
             if remaining <= 0 then
                  local t = Translations[SelectedLanguage] or Translations.English
                TimerLabel.Text = t.KeyExpired
                IsLoggedIn = false
                KeyInput.TextEditable = true
                 if isfile(FileName) then delfile(FileName) end
                   break
            end
            TimerLabel.Text = FormatTime(remaining)
        end
    end)
end

if RequireKeyForEU then
    if isfile(FileName) then
        local savedTime = tonumber(readfile(FileName))
        if savedTime and savedTime > os.time() then
            IsLoggedIn = true
            EndTime = savedTime
            KeyInput.TextEditable = false
        else
            delfile(FileName)
        end
    end
else
    IsLoggedIn = true 
    EndTime = math.huge 
end

local function FadeOutAndExecute(url)
    MuteManager:FadeInSounds(1) 
    
    if IntroMusic and IntroMusic.Parent then
        local musicFadeOut = TweenService:Create(IntroMusic, TweenInfo.new(1, Enum.EasingStyle.Linear), {Volume = 0})
         musicFadeOut:Play()
        task.delay(1, function()
            if IntroMusic and IntroMusic.Parent then
                IntroMusic:Stop()
                IntroMusic:Destroy()
            end
        end)
    end
    
    MainFrame.ClipsDescendants = true 
    local ti = TweenInfo.new(0.6, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
    
    SafeTween(MainFrame, ti, {
        Size = UDim2.new(0, 0, 0, 0),
        Position = UDim2.new(0.5, 0, 0.5, 0)
    })
    
    for _, element in pairs(ContentFrame:GetDescendants()) do
        if element:IsA("TextLabel") or element:IsA("TextBox") or element:IsA("TextButton") then
             SafeTween(element, ti, {TextTransparency = 1, BackgroundTransparency = 1})
         elseif element:IsA("ImageLabel") or element:IsA("ImageButton") then
            SafeTween(element, ti, {ImageTransparency = 1})
        elseif element:IsA("Frame") then
            SafeTween(element, ti, {BackgroundTransparency = 1})
        end
    end
    SafeTween(GameDetailPanel, ti, {BackgroundTransparency = 1})
    
    SafeTween(BackgroundOverlay, ti, {BackgroundTransparency = 1})
    task.wait(0.6)
    ScreenGui:Destroy()
    if url then loadstring(game:HttpGet(url))() end
end

local function CreateGameCard(name, url, assetId, order, detailId)
    local container = Instance.new("Frame")
    container.Size = UDim2.new(0, 65, 0, 49)
    container.BackgroundTransparency = 1
    container.BorderSizePixel = 0
    container.LayoutOrder = order
    container.Parent = GameScrollFrame
    container.ZIndex = 12

    local imgBtn = Instance.new("ImageButton")
    imgBtn.Parent = container
    imgBtn.Size = UDim2.new(1, 0, 1, 0)
    imgBtn.Position = UDim2.new(0, 0, 0, 0)
    imgBtn.BackgroundTransparency = 1
    imgBtn.Image = "rbxthumb://type=Asset&id=" .. assetId .. "&w=150&h=150"
    imgBtn.ImageTransparency = 1
    imgBtn.ScaleType = Enum.ScaleType.Stretch
    imgBtn.ZIndex = 13
    Instance.new("UICorner", imgBtn).CornerRadius = UDim.new(0, 6)
    
    local imgScale = Instance.new("UIScale", imgBtn)
    imgScale.Scale = 1

    imgBtn:SetAttribute("TargetTrans", 1)

    -- النص ابن للصورة نفسها (imgBtn) وليس للـ container، فيتحرك معها تلقائياً
    -- في أي تحريك (مثل نظام التبعيد عند الضغط) لأن موضعه نسبي لأبيه الصورة
    local lbl = Instance.new("TextLabel")
    lbl.Parent = imgBtn
    lbl.Size = UDim2.new(1, -6, 0, 14)
    lbl.Position = UDim2.new(0, 3, 0, 2)
    lbl.BackgroundTransparency = 1
    lbl.Font = Enum.Font.Gotham
    lbl.Text = name
    lbl.TextColor3 = Color3.fromRGB(255, 255, 255)
    lbl.TextSize = 7
    lbl.TextTransparency = 1
    lbl.TextWrapped = false
    lbl.TextScaled = false
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.TextYAlignment = Enum.TextYAlignment.Top
    lbl.ZIndex = 15

    -- تسجيل صورة هذا الكرت في القائمة المركزية لتحميلها ضمن شاشة
    -- التحميل السوداء عند بدء التشغيل. أي كرت يُنشأ مستقبلاً (حتى لو
    -- أُضيف بعد هذا التحديث) سينضم تلقائياً لأن RegisterPreloadAsset
    -- تُستدعى هنا لكل كرت يُنشأ، بلا حاجة لتعديل كود الـ Intro.
    RegisterPreloadAsset(imgBtn)

    -- [إصلاح]: الصورة الكبيرة في لوحة التفاصيل (DetailImage، مقاس 420x420)
    -- لم تكن تُحمَّل إطلاقاً حتى أول ضغطة فعلية على الكرت، فكانت تظهر
    -- بشكل متقطع (لأنها تُطلب من سيرفرات Roblox في نفس لحظة الضغطة) ولا
    -- تصبح سلسة إلا في الضغطة الثانية بعد أن تُخزَّن مؤقتاً. لحل هذا
    -- نخلق ImageLabel وهمياً (غير مرئي ومنفصل عن الشجرة الظاهرة) بنفس
    -- معرّف الصورة الكبيرة ونسجّله في نفس قائمة التحميل المركزية، فيتم
    -- تحميل وتخزين هذه الصورة أيضاً أثناء شاشة التحميل السوداء قبل أن
    -- يضغط المستخدم على أي كرت إطلاقاً.
    if detailId and detailId ~= "" then
        local detailPreloadImg = Instance.new("ImageLabel")
        detailPreloadImg.Name = "DetailPreload_" .. name
        detailPreloadImg.Visible = false
        detailPreloadImg.Size = UDim2.new(0, 1, 0, 1)
        detailPreloadImg.BackgroundTransparency = 1
        detailPreloadImg.Image = "rbxthumb://type=Asset&id=" .. detailId .. "&w=420&h=420"
        detailPreloadImg.Parent = CoreGui
        RegisterPreloadAsset(detailPreloadImg)
    end

    -- ملاحظة: التحميل يتم مركزياً عبر _PendingPreloadAssets في شاشة التحميل السوداء.
-- لو أُنشئ الكرت بعد اختفاء الشاشة، ContentProvider يخزّن الصور تلقائياً عند أول عرض.
AnimateButton(imgBtn, function()
        if IsLoggedIn then
            SelectGame(imgBtn, name, url, assetId, detailId)
        end
    end)

    table.insert(GameImagesList, imgBtn)
    table.insert(GameLabelsList, lbl)
end

local function OpenDinoPanel()
    -- [إصلاح]: نضع _DinoOpening = true دائماً بغض النظر عن نوع الإدخال (PC أو Touch)
    -- لمنع أي توين hover/leave من التدخل أثناء توين الإخفاء
    _DinoOpening = true

    -- [إصلاح]: كانت EasingDirection.In تجعل التلاشي يبدو بطيئاً جداً معظم الوقت
    -- ثم يقفز بسرعة في آخر جزء من الزمن (يظهر فجأة). تم تغييرها لـ Out
    -- لتطابق نفس أسلوب التلاشي السلس المستخدم في CloseDinoPanel وباقي الصفحة
    local ti = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)

    -- [إصلاح]: إلغاء أي توين hover/press نشط على LaunchBtn قبل الإخفاء
    -- لأن AnimateButton يشغّل tweens مستقلة تتعارض مع SafeTween أدناه وتمنع الإخفاء السلس
    do
        local existing = _ActiveTweensByInstance[LaunchBtn]
        if existing then
            for propName, t in pairs(existing) do
                if t then t:Cancel() end
                existing[propName] = nil
            end
        end
        local launchScale = LaunchBtn:FindFirstChildOfClass("UIScale")
        if launchScale then
            local existingScale = _ActiveTweensByInstance[launchScale]
            if existingScale then
                for propName, t in pairs(existingScale) do
                    if t then t:Cancel() end
                    existingScale[propName] = nil
                end
            end
        end
    end

    -- إخفاء صفحة الألعاب أولاً بنفس أسلوب FadeOutPage("Games")، ثم تبديل الصفحة فعلياً
    local tiOut = TweenInfo.new(0.25, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    SafeTween(GamesContentBox, tiOut, {BackgroundTransparency = 1})
    SafeTween(GameDetailPanel, tiOut, {BackgroundTransparency = 1})

    SafeTween(DetailNameLabel, tiOut, {TextTransparency = 1})
    SafeTween(LaunchBtn,       tiOut, {BackgroundTransparency = 1, TextTransparency = 1})
    SafeTween(AntiBanLabel, tiOut, {TextTransparency = 1})
    SafeTween(GameScrollFrame, tiOut, {ScrollBarImageTransparency = 1})
    for _, img in pairs(GameImagesList) do
        SafeTween(img, tiOut, {ImageTransparency = 1})
    end
    for _, lbl in pairs(GameLabelsList) do
        SafeTween(lbl, tiOut, {TextTransparency = 1})
    end
    SafeTween(WarningLabel, tiOut, {TextTransparency = 1})

    task.wait(0.25)
    GameSelectionFrame.Visible = false
    _DinoOpening = false  -- انتهى الإخفاء، نعيد تفعيل hover/press بشكل طبيعي

    -- إظهار صفحة Dino كصفحة مستقلة (نفس منطقة المحتوى، بلا تغطية للعنوان أو شريط التنقل)
    DinoLaunchPanel.Visible = true
    PlayPageChangeSound()
    DinoRunBtn.BackgroundColor3 = TargetGlowColor
    DinoCloseBtn.BackgroundColor3 = TargetGlowColor
    DinoTextScroll.ScrollBarImageColor3 = TargetGlowColor
    DinoPanelBg.ScrollBarImageColor3 = TargetGlowColor

    DinoPanelImage.Image = "rbxthumb://type=Asset&id=" .. (_DinoPanelImageId or "133578398288699") .. "&w=420&h=420"

local function SetupDinoBtn(btn, onClickCb)
    local scale = btn:FindFirstChildOfClass("UIScale") or Instance.new("UIScale", btn)
    local tiHover = TweenInfo.new(0.14, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    local tiPress = TweenInfo.new(0.1,  Enum.EasingStyle.Sine, Enum.EasingDirection.Out)

    btn.MouseEnter:Connect(function()
        SafeTween(scale, tiHover, {Scale = 1.06})
        if btn.BackgroundTransparency < 1 then
            SafeTween(btn, tiHover, {BackgroundTransparency = btn.BackgroundTransparency - 0.07})
        end
    end)

    btn.MouseLeave:Connect(function()
        SafeTween(scale, tiHover, {Scale = 1.0})
        if btn == DinoRunBtn then
            SafeTween(btn, tiHover, {BackgroundTransparency = 0.9})
        elseif btn == DinoCloseBtn then
            SafeTween(btn, tiHover, {BackgroundTransparency = 0.9})
        end
    end)

    btn.MouseButton1Down:Connect(function()
        SafeTween(scale, tiPress, {Scale = 0.93})
        if btn.BackgroundTransparency < 1 then
            SafeTween(btn, tiPress, {BackgroundTransparency = math.min(1, btn.BackgroundTransparency + 0.12)})
        end
    end)

    btn.MouseButton1Click:Connect(function()
        PlayClickSound()
        SafeTween(scale, TweenInfo.new(0.18, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1.0})
        if onClickCb then onClickCb() end
    end)
end

local function SwitchSettingTab(targetIndex)
    if CurrentSettingTab == targetIndex then return end
    
    local ti = TweenInfo.new(0.25, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
    
    for _, el in ipairs(SettingsContentBox:GetDescendants()) do
        if el:IsA("TextLabel") or el:IsA("TextButton") then
            SafeTween(el, ti, {TextTransparency = 1, BackgroundTransparency = 1})
        elseif el:IsA("ImageLabel") then
            SafeTween(el, ti, {ImageTransparency = 1})
        elseif el:IsA("Frame") and el ~= SettingsContentBox then
             SafeTween(el, ti, {BackgroundTransparency = 1})
        elseif el:IsA("UIStroke") then
             SafeTween(el, ti, {Transparency = 1})
        end
    end
    
    SafeTween(LangVerticalScroll, ti, {ScrollBarImageTransparency = 1})
    SafeTween(UpdatesScroll, ti, {ScrollBarImageTransparency = 1})

local function FadeIn()
    while not _IntroReadyToLoad do
        task.wait()
    end

    -- ====================================================================
    -- مرحلة 1: الخلفية الملونة تظهر فوراً (تحت الشاشة السوداء)
    -- ====================================================================
    BackgroundOverlay.BackgroundTransparency = 0.4
    MainFrame.BackgroundTransparency = 0

    -- الـ glow يبدأ فوراً
    task.spawn(function()
        local start = os.clock()
        local duration = 2.0
        while os.clock() - start < duration do
            local pct = (os.clock() - start) / duration
            GlowAlpha = pct
            task.wait()
        end
        GlowAlpha = 1
    end)

    -- ====================================================================
    -- مرحلة 2: بعد ثانيتين الشاشة السوداء تروح، الأنيميشن يظهر
    -- ====================================================================
    MuteManager:Init()
    MuteManager:FadeOutSounds(1)

    task.wait(2)

    local fadeOut = TweenService:Create(
        IntroFrame,
        TweenInfo.new(1.0, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
        {BackgroundTransparency = 1}
    )
    fadeOut:Play()

    -- الأغنية تشتغل مع ظهور الأنيميشن
    IntroMusic:Play()
    SafeTween(IntroMusic, TweenInfo.new(1.5, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {Volume = 0.3})

    -- نص التحميل يظهر في أسفل اليسار أثناء الأنيميشن
    -- ====================================================================
    -- Acrocan0 Hub + النقاط + نص التحميل: كلهم داخل frame واحد
    -- Frame موقعه نسبي (أسفل اليسار) فيتكيف مع أي حجم شاشة
    -- ====================================================================

    -- حجم النصوص مبني على SizeIndex (نفس إعداد الحجم في الـ Settings)
    -- Base: SizeIndex=15 (default) → TextSize 24/18، يتغير مع كل خطوة
    local function GetIntroTextScale()
        return ScaleValues[SizeIndex] / ScaleValues[15]
    end
    local baseTitleSize  = 24
    local baseLoadSize   = 18
    local baseTitleW     = 160
    local baseTitleH     = 28
    local baseDotsX      = 14 + 160 + 6

    -- نحسب الـ Y positions بشكل ديناميكي بناءً على حجم النص الفعلي
    -- بدل قيم ثابتة تتعطل لما يتغير الحجم
    local baseDotSize    = 3
    local baseDotSpacing = 8
    local introScale     = GetIntroTextScale()
    local titleSize      = math.round(baseTitleSize * introScale)
    local loadSize       = math.round(baseLoadSize  * introScale)
    local dotSize        = math.max(1, math.round(3 * introScale) - math.round(2 * math.abs(1 - introScale)))
    local dotSpacing     = math.round(baseDotSpacing * introScale)
    local realTextW = 0
    pcall(function()
        local ts = game:GetService("TextService")
        local v2 = ts:GetTextSize("Acrocan0 Hub", titleSize, Enum.Font.GothamBold, Vector2.new(9999, 9999))
        realTextW = v2.X
    end)
    if realTextW == 0 then realTextW = math.round(baseTitleW * introScale) end
    local dotsStartX = 14 + realTextW + 5

    -- المسافة من أسفل الـ frame: نص صغير تحت + عنوان فوقه + مسافة بسيطة
    local bottomPad  = math.round(10 * introScale)
    local loadLineH  = loadSize + 4
    local titleLineH = titleSize + 4
    local baseTitleY = -(bottomPad + loadLineH + titleLineH)   -- Y للعنوان
    local baseLoadY  = -(bottomPad + loadLineH)                 -- Y لنص التحميل
    local baseDotsY  = -(loadLineH + titleLineH) - 6 + math.floor(titleLineH * 0.1) - 10

    local IntroHUDFrame = Instance.new("Frame")
    IntroHUDFrame.Parent = ScreenGui
    IntroHUDFrame.BackgroundTransparency = 1
    IntroHUDFrame.AnchorPoint = Vector2.new(0, 1)
    IntroHUDFrame.Position = UDim2.new(0, 0, 1, 0)
    IntroHUDFrame.Size = UDim2.new(0.5, 0, 0, 80)
    IntroHUDFrame.ZIndex = 999

    -- UIScale مربوط بحجم الشاشة مثل باقي الواجهة
    local IntroHUDScale = Instance.new("UIScale")
    IntroHUDScale.Scale = GetBackgroundScale()
    IntroHUDScale.Parent = IntroHUDFrame
    Camera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
        if IntroHUDScale and IntroHUDScale.Parent then
            IntroHUDScale.Scale = GetBackgroundScale()
        end
    end)

    -- "Acrocan0 Hub"
    local HubTitleLabel = Instance.new("TextLabel")
    HubTitleLabel.Parent = IntroHUDFrame
    HubTitleLabel.BackgroundTransparency = 1
    HubTitleLabel.AnchorPoint = Vector2.new(0, 1)
    HubTitleLabel.Position = UDim2.new(0, 14, 1, baseTitleY)
    HubTitleLabel.Size = UDim2.new(0, baseTitleW, 0, baseTitleH)
    HubTitleLabel.Font = Enum.Font.GothamBold
    HubTitleLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    HubTitleLabel.TextSize = titleSize
    HubTitleLabel.TextXAlignment = Enum.TextXAlignment.Left
    HubTitleLabel.TextTransparency = 1
    HubTitleLabel.ZIndex = 999
    HubTitleLabel.Text = "Acrocan0 Hub"

    -- النقاط الثلاث داخل الـ frame، جنب العنوان

    local Dot1 = Instance.new("Frame")
    Dot1.Parent = IntroHUDFrame
    Dot1.BackgroundColor3 = TargetGlowColor:Lerp(Color3.new(1,1,1), 0.75)
    Dot1.BackgroundTransparency = 1
    Dot1.AnchorPoint = Vector2.new(0, 1)
    Dot1.Position = UDim2.new(0, dotsStartX, 1, baseDotsY)
    Dot1.Size = UDim2.new(0, dotSize, 0, dotSize)
    Dot1.BorderSizePixel = 0
    Dot1.ZIndex = 999
    Instance.new("UICorner", Dot1).CornerRadius = UDim.new(1, 0)

    local Dot2 = Instance.new("Frame")
    Dot2.Parent = IntroHUDFrame
    Dot2.BackgroundColor3 = TargetGlowColor:Lerp(Color3.new(1,1,1), 0.75)
    Dot2.BackgroundTransparency = 1
    Dot2.AnchorPoint = Vector2.new(0, 1)
    Dot2.Position = UDim2.new(0, dotsStartX + dotSpacing, 1, baseDotsY)
    Dot2.Size = UDim2.new(0, dotSize, 0, dotSize)
    Dot2.BorderSizePixel = 0
    Dot2.ZIndex = 999
    Instance.new("UICorner", Dot2).CornerRadius = UDim.new(1, 0)

    local Dot3 = Instance.new("Frame")
    Dot3.Parent = IntroHUDFrame
    Dot3.BackgroundColor3 = TargetGlowColor:Lerp(Color3.new(1,1,1), 0.75)
    Dot3.BackgroundTransparency = 1
    Dot3.AnchorPoint = Vector2.new(0, 1)
    Dot3.Position = UDim2.new(0, dotsStartX + dotSpacing * 2, 1, baseDotsY)
    Dot3.Size = UDim2.new(0, dotSize, 0, dotSize)
    Dot3.BorderSizePixel = 0
    Dot3.ZIndex = 999
    Instance.new("UICorner", Dot3).CornerRadius = UDim.new(1, 0)

    -- نص التحميل تحت العنوان
    local LoadingLabel = Instance.new("TextLabel")
    LoadingLabel.Parent = IntroHUDFrame
    LoadingLabel.BackgroundTransparency = 1
    LoadingLabel.AnchorPoint = Vector2.new(0, 1)
