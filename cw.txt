function setCoreGuiState(state)
    if state == false then
        for _, child in ipairs(CoreGui:GetChildren()) do
            if child.Name ~= "RobloxGui" then
                coreBackup[child] = child.Parent
                child.Parent = nil
            end
        end
        for _, gui in ipairs(PlayerGui:GetChildren()) do
            if gui:IsA("ScreenGui") and gui.Name ~= "Acrocano_LoadingWall" then
                playerGuiBackup[gui] = gui.Parent
                gui.Parent = nil
            end
        end
        pcall(function() StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.All, false) end)
    else
        for child, parent in pairs(coreBackup) do
            if child then pcall(function() child.Parent = parent end) end
        end
        for gui, parent in pairs(playerGuiBackup) do
            if gui then pcall(function() gui.Parent = parent end) end
        end
        coreBackup = {}
        playerGuiBackup = {}
        pcall(function()
            StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.All, true)
            StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.Backpack, false)
        end)
    end
end

function lockCamera(roomPos)
    local camera = workspace.CurrentCamera
    if not camera then return end
    camera.CameraType = Enum.CameraType.Scriptable
    lp.CameraMaxZoomDistance = 0.5
    lp.CameraMinZoomDistance = 0.5
    local startTime = tick()
    cameraConnection = RunService.RenderStepped:Connect(function()
        local time = tick() - startTime
        local shakeX = math.sin(time*0.5)*0.02
        local shakeY = math.cos(time*0.4)*0.02
        local camPos = roomPos + Vector3.new(0,10,-2)
        local lookAt = roomPos + Vector3.new(0,10,10)
        camera.CFrame = CFrame.new(camPos, lookAt) * CFrame.Angles(shakeY,shakeX,0)
    end)
end

function unlockCamera()
    if cameraConnection then cameraConnection:Disconnect() cameraConnection = nil end
    local camera = workspace.CurrentCamera
    if camera then
        camera.CameraType = Enum.CameraType.Custom
    end
    lp.CameraMaxZoomDistance = 128
    lp.CameraMinZoomDistance = 0.5
    setCoreGuiState(true)
end

local function SetMoveVector(vec)
    pcall(function()
        if not Controls then Controls = require(lp:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule")):GetControls() end
        local activeController = Controls:GetActiveController()

local function ApplyJoystickSettings()
    if GrayGui and GrayGui:FindFirstChild("TouchBlocker") then
        local main = GrayGui.TouchBlocker.MainFrame
        main.Size = UDim2.new(0, _G_settings.JoySize, 0, _G_settings.JoySize)
        main.Position = UDim2.new(0, _G_settings.JoyPosX, 0.5, _G_settings.JoyPosY)
        local stick = main.Stick
        local half = _G_settings.JoySize / 2
        stick.Size = UDim2.new(0, half, 0, half)
        if not activeTouch then
            stick.Position = UDim2.new(0.5, -half/2, 0.5, -half/2)
        end
    end
end
local function stopMoving()
    activeTouch = nil
    moveDir = Vector3.new(0,0,0)
    if GrayGui and GrayGui:FindFirstChild("TouchBlocker") then
        local stick = GrayGui.TouchBlocker.MainFrame.Stick
        local half = _G_settings.JoySize / 2
        stick:TweenPosition(UDim2.new(0.5, -half/2, 0.5, -half/2), "Out", "Back", 0.1)
    end
    SetMoveVector(Vector3.new(0,0,0))
end
local function setupGrayJoy()
    local sg = Instance.new("ScreenGui", pgui)
    sg.Name = "Gray_Movement"
    sg.DisplayOrder = -100
    sg.Enabled = false
    sg.ResetOnSpawn = false
    local touchBlocker = Instance.new("Frame", sg)
    touchBlocker.Name = "TouchBlocker"
    touchBlocker.Size = UDim2.new(0,400,0,180)
    touchBlocker.Position = UDim2.new(0,10,1,-200)
    touchBlocker.BackgroundTransparency = 1
    touchBlocker.Active = true
    local main = Instance.new("Frame", touchBlocker)
    main.Name = "MainFrame"
    main.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    main.BackgroundTransparency = 0.80
    main.Active = true
    main:SetAttribute("IsJoystick", true)
    Instance.new("UICorner", main).CornerRadius = UDim.new(1,0)
    local stick = Instance.new("Frame", main)
    stick.Name = "Stick"
    stick.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    stick.BackgroundTransparency = 0.80
    stick:SetAttribute("IsJoystick", true)
    Instance.new("UICorner", stick).CornerRadius = UDim.new(1,0)
    GrayGui = sg
    ApplyJoystickSettings()
    main.InputBegan:Connect(function(input)
        if (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1) then
            activeTouch = input
            local center = main.AbsolutePosition + (main.AbsoluteSize/2)
            local diff = Vector2.new(input.Position.X - center.X, input.Position.Y - center.Y)
            local limit = main.AbsoluteSize.X/2
            local dist = math.min(diff.Magnitude, limit)
            if dist > 0.05 then
                local dir = diff.Unit
                SetMoveVector(Vector3.new(dir.X*(dist/limit), 0, dir.Y*(dist/limit)))
            end
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == activeTouch and sg.Enabled then
            local center = main.AbsolutePosition + (main.AbsoluteSize/2)
            local diff = Vector2.new(input.Position.X - center.X, input.Position.Y - center.Y)
            local limit = main.AbsoluteSize.X/2
            local dist = math.min(diff.Magnitude, limit)
            local dir = diff.Unit
            local half = _G_settings.JoySize/2
            stick.Position = UDim2.new(0.5, (dir.X*dist)-half/2, 0.5, (dir.Y*dist)-half/2)
            if dist > 0.05 then SetMoveVector(Vector3.new(dir.X*(dist/limit), 0, dir.Y*(dist/limit))) else SetMoveVector(Vector3.new(0,0,0)) end
        end
    end)
    UserInputService.InputEnded:Connect(function(input) if input == activeTouch then stopMoving() end end)
    UserInputService.WindowFocusReleased:Connect(stopMoving)
    return sg
end
local function UpdateGrayMovementState()
    local shouldBeActive = _G_settings.CustomJoystick
    local touchGui = pgui and pgui:FindFirstChild("TouchGui")
    local function toggleDefaultThumbstick(visible)
        if touchGui then
            local frame = touchGui:FindFirstChild("TouchControlFrame")
            if frame then
                for _, v in pairs(frame:GetChildren()) do
                    if string.find(v.Name, "Thumbstick") then
                        for _, desc in pairs(v:GetDescendants()) do if desc:IsA("ImageLabel") or desc:IsA("ImageButton") then desc.ImageTransparency = visible and 0 or 1 end end
                        v.BackgroundTransparency = 1
                    end
                end
            end
        end
    end
    if shouldBeActive then
        toggleDefaultThumbstick(false)
        if not GrayGui then
            GrayGui = setupGrayJoy() -- setupGrayJoy تستدعي ApplyJoystickSettings بالداخل
        else
            GrayGui.Enabled = true
            ApplyJoystickSettings()
        end
        if GrayGui then GrayGui.Enabled = true end
    else
        toggleDefaultThumbstick(true)
        if GrayGui and GrayGui.Enabled then GrayGui.Enabled = false; stopMoving() end
    end
end

        clearVisualPath()
        playerLinePart = nil
        startIndex = startIndex or 1
        if not waypoints or #waypoints < 2 then return end
        -- ارسم الـ parts الثابتة بين الـ waypoints
        for i = startIndex, #waypoints-1 do
            local startPos = waypoints[i].Position
            local endPos = waypoints[i+1].Position
            local distance = (startPos-endPos).Magnitude
            if distance > 0.1 then
                local line = Instance.new("Part")
                line.Name = "PathLine"
                line.Anchored = true; line.CanCollide = false; line.CanTouch = false; line.CanQuery = false
                line.Material = Enum.Material.Neon; line.Color = Color3.fromRGB(0,255,128); line.Transparency = 0.6
                line.Size = Vector3.new(0.4,0.05,distance)
                line.CFrame = CFrame.lookAt((startPos+endPos)/2, endPos)
                line.Parent = visualFolder
            end
        end
        -- اصنع الـ part المتحرك بين اللاعب والـ waypoint الأول
        local dynLine = Instance.new("Part")
        dynLine.Name = "PlayerPathLine"

        if not playerLinePart or not playerLinePart.Parent then return end
        if not currentWaypoints or #currentWaypoints < 1 then return end
        local rootPart = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
        if not rootPart then return end
        local idx = math.min(currentWaypointIndex, #currentWaypoints)
        local fromPos = Vector3.new(rootPart.Position.X, currentWaypoints[idx].Position.Y, rootPart.Position.Z)
        local toPos = currentWaypoints[idx].Position
        local dist = (fromPos - toPos).Magnitude
        if dist > 0.2 then
            playerLinePart.Size = Vector3.new(0.4, 0.05, dist)
            playerLinePart.CFrame = CFrame.lookAt((fromPos+toPos)/2, toPos)
            playerLinePart.Transparency = 0.6
        else
            playerLinePart.Transparency = 1
        end
    end

    local function getNearestEnemy()
        local character = lp.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return nil end
        local playerPos = character.HumanoidRootPart.Position
        local nearestEnemy = nil
        local shortestDistance = math.huge
        for _, p in ipairs(game.Players:GetPlayers()) do
            if p ~= lp and p.Character and p.Character:FindFirstChild("Humanoid") and p.Character:FindFirstChild("HumanoidRootPart") then
                if p.Character.Humanoid.Health > 0 and not _G_settings.IgnoredPlayers[p.Name] then
                    local distance = getHorizontalDistance(playerPos, p.Character.HumanoidRootPart.Position)
                    if distance < shortestDistance then shortestDistance = distance; nearestEnemy = p.Character end
                end
            end
        end
        return nearestEnemy
    end

                    character = lp.Character

                    local wp = currentWaypoints[currentWaypointIndex]
                    humanoid.AutoRotate = false
                    -- أعد حساب الـ path كل ثانية
                    if tick() - lastRecompute >= 1 then
                        lastRecompute = tick()
                        local success2, _ = pcall(function() path:ComputeAsync(rootPart.Position, targetEnemy.HumanoidRootPart.Position) end)
                        if success2 and path.Status == Enum.PathStatus.Success then
                            currentWaypoints = path:GetWaypoints()

                                if dist > 2 then currentWaypointIndex = i; break end
                            end
                        end

                    if hum then hum.AutoRotate = true end

    ["utility5Proxy"] = "Grenade",

local MasterAntiListener = nil
local function SetupMasterListener()
    if MasterAntiListener then return end
    MasterAntiListener = workspace.DescendantAdded:Connect(function(obj)
        -- [OPTIMIZED] Early exit to save Memory/CPU if features are disabled
        if not (_G_settings.AntiBearTrap or _G_settings.AntiGrenade or _G_settings.AntiC4 or _G_settings.AntiMolotov) then return end
        if not obj:IsA("BasePart") then return end 
        
        local name = obj.Name
        local hazardType = HazardLookup[name]
        
        if hazardType == "BearTrap" and _G_settings.AntiBearTrap then
            task.delay(0.1, function()
                if obj.Parent and (obj.Parent.Name == "OpenBearTrap" or obj.Parent.Parent.Name == "EffectsJunk") then obj:Destroy() end
            end)
        elseif hazardType == "Grenade" and _G_settings.AntiGrenade then
            task.delay(2, function()
                if obj and obj.Parent then ActiveHazards.Grenades[obj] = { shieldRadius = 20 } end
            end)
        elseif hazardType == "C4" and _G_settings.AntiC4 then
            ActiveHazards.C4s[obj] = { shieldRadius = 24 }
        elseif _G_settings.AntiMolotov and (obj:GetAttribute("MolotovId") or (name == "" and obj:GetAttribute("DamagePerSecond"))) then
            if not obj:GetAttribute("Acrocan0_ShieldActive") then
                obj:SetAttribute("Acrocan0_ShieldActive", true)
                ActiveHazards.Molotovs[obj] = { size = 35, addedTick = tick() }
            end
        end
    end)
end

    local function calculatePredictedDistanceSq(targetRoot, myPosX,myPosY,myPosZ, ping)
        local targetPos = targetRoot.Position
        local targetVel = targetRoot.AssemblyLinearVelocity
        -- نحد تأثير الـ velocity عشان ما يضخم المسافة غلط لما اللاعب يقفز أو يركض
        local vx = math.clamp(targetVel.X * ping, -5, 5)
        local vy = math.clamp(targetVel.Y * ping, -3, 3)
        local vz = math.clamp(targetVel.Z * ping, -5, 5)
        local dx = (targetPos.X + vx) - myPosX
        local dy = (targetPos.Y + vy) - myPosY
        local dz = (targetPos.Z + vz) - myPosZ
        return dx*dx + dy*dy + dz*dz
    end

    local function rawDistanceSq(targetRoot, myPosX, myPosY, myPosZ)
        local p = targetRoot.Position
        local dx, dy, dz = p.X - myPosX, p.Y - myPosY, p.Z - myPosZ
        return dx*dx + dy*dy + dz*dz
    end
    
    local function setupParrySystems(playerToMonitor, char)
        local humanoid = char:WaitForChild("Humanoid",5)
        if not humanoid then return end
        local animator = humanoid:WaitForChild("Animator",5) or humanoid
        if not animator then return end
        animator.AnimationPlayed:Connect(function(track)
            if not _G_settings.AutoParrySystem then return end
            if myRoot then
                local targetRoot = rootCache[playerToMonitor]
                if targetRoot then
                    local mp = myRoot.Position
                    if rawDistanceSq(targetRoot, mp.X, mp.Y, mp.Z) > (60*60) then return end
                end
            end
            
            local animObj = track.Animation
            if not animObj then return end

            -- FIX 1: جلب الاسم من المصادر الصح (حل مشكلة الفراغ)
            local tn = track.Name or ""
            local an = animObj.Name or ""
            local localAnimName = string_lower(
                (tn ~= "" and tn ~= "AnimationTrack") and tn or
                (an ~= "" and an ~= "Animation") and an or ""
            )
            local isAttack, isDefense = classifyAnimation(localAnimName)

            if not isAttack and not isDefense then
                local animId = animObj.AnimationId
                if animId then
                    local idNum = idNumCache[animId]
                    if not idNum then
                        idNum = tonumber(string_match(animId, "%d+")) or 0
                        if idNum ~= 0 then idNumCache[animId] = idNum end
                    end
                    if idNum ~= 0 then
                        local name = animationNameCache[idNum]
                        -- FIX 2 & 3: لما الاسم موجود في الكاش نصنفه، لما مو موجود نطلبه
                        -- ونشغل الباري مباشرة بدل تجاهل الضربة (حل تجاهل الضربات والذاكرة)
                        if name then
                            isAttack, isDefense = classifyAnimation(name)
                        else
                            getAnimInfoCache(idNum)
                        end
                    end
                end
            end
            if isAttack or isDefense then
                if myRoot then
                    local targetRoot = rootCache[playerToMonitor]
                    if targetRoot then
                        local myPos = myRoot.Position
                        -- نستخدم المسافة الحقيقية للفحص الفوري (أدق وأسرع)
                        local distSq = rawDistanceSq(targetRoot, myPos.X, myPos.Y, myPos.Z)
                        if isAttack and config.AutoParry and distSq <= config.AutoParryDistSq then performParry(); return
                        elseif isDefense and config.AntiParry and distSq <= config.AntiParryDistSq then triggerAntiParry(); return end
                    end
                end
                activeAnimations[track] = { Player = playerToMonitor, IsAttack = isAttack, IsDefense = isDefense, Handled = false, TimeStarted = os_clock() }
            end
        end)
    end

        player.CharacterAdded:Connect(onCharAdded)
        player.CharacterRemoving:Connect(function() rootCache[player] = nil end)
        if player.Character then task_defer(onCharAdded, player.Character) end
    end
    Players.PlayerRemoving:Connect(function(player)
        rootCache[player] = nil
        for track, info in pairs(activeAnimations) do if info.Player == player then activeAnimations[track]=nil end end
    end)
    for _, player in ipairs(Players:GetPlayers()) do monitorPlayer(player) end
    Players.PlayerAdded:Connect(monitorPlayer)
    
    local lastParryProcess = 0
    local CLOSE_DIST_SQ = 10*10 -- أقل من 10 studs = قريب جداً، استجابة فورية
    RunService.Heartbeat:Connect(function()
        if not _G_settings.AutoParrySystem or not myRoot then return end

        local now = tick()
        local myPos = myRoot.Position
        local mpx,mpy,mpz = myPos.X, myPos.Y, myPos.Z
        local currentTime = os_clock()

        -- فحص سريع: إذا في أنيميشن نشط وصاحبه قريب جداً → استجب فوراً بدون interval
        local hasCloseTarget = false
        for track, info in pairs(activeAnimations) do
            if not info.Handled and track.IsPlaying then
                local targetRoot = rootCache[info.Player]
                if targetRoot then
                    if rawDistanceSq(targetRoot, mpx, mpy, mpz) <= CLOSE_DIST_SQ then
                        hasCloseTarget = true
                        break
                    end
                end
            end
        end

        -- إذا مافي هدف قريب نطبق الـ interval العادي
        if not hasCloseTarget and now - lastParryProcess < 0.05 then return end
        lastParryProcess = now

        local currentPing = getPing()
        for track, info in pairs(activeAnimations) do
            if info.Handled or not track.IsPlaying or (currentTime - info.TimeStarted) > config.AnimationExpiry then
                activeAnimations[track] = nil
            else
                local targetRoot = rootCache[info.Player]
                if targetRoot then
                    local realDistSq = rawDistanceSq(targetRoot, mpx, mpy, mpz)
                    local predDistSq = calculatePredictedDistanceSq(targetRoot, mpx, mpy, mpz, currentPing)
                    local distSq = math.min(realDistSq, predDistSq)
                    if info.IsAttack and config.AutoParry and distSq <= config.AutoParryDistSq then
                        info.Handled = true; performParry()
                    elseif info.IsDefense and config.AntiParry and distSq <= config.AntiParryDistSq then
                        info.Handled = true; triggerAntiParry()
                    end
                else activeAnimations[track] = nil end
            end
        end
    end)
end

local isHubOpen = false

task.spawn(function()
    local timeCounter = 0
    while true do
        if isHubOpen then
            timeCounter = timeCounter + 0.007
            CurrentGlowColor = CurrentGlowColor:Lerp(TargetGlowColor, 0.03)
            for _, data in pairs(LayersData) do
                if data.Instance and data.Instance.Parent then
                    local offsetX, offsetY, sizePulse
                    if CurrentAnimStyle == "1v" then
                        local waveAngle = timeCounter * 0.6 + (data.OffsetFactor * math.pi * 1.5)
                        offsetX = math.sin(waveAngle) * 120
                        offsetY = math.cos(waveAngle * 0.5) * 40
                        sizePulse = 1 + (math.sin(waveAngle * 0.8) * 0.12)
                    elseif CurrentAnimStyle == "2v" then
                        local waveAngle = timeCounter * 1.2 + (data.OffsetFactor * math.pi * 2.5)
                        offsetX = math.sin(waveAngle) * 80
                        offsetY = math.cos(waveAngle * 0.8) * 80
                        sizePulse = 1 + (math.sin(timeCounter * 1.5 + data.OffsetFactor) * 0.25)
                    elseif CurrentAnimStyle == "3v" then
                        local speed = 0.15 + (data.OffsetFactor * 0.35)
                        local seed = data.OffsetFactor * 100
                        local layerAngle = timeCounter * (speed * 1) + seed
                        offsetX = (math.sin(layerAngle) * 40) - 50
                        offsetY = math.cos(layerAngle * 0.8) * 20
                        sizePulse = 1 + (math.sin(layerAngle * 1.2) * 0.25)
                    end
                    local distanceFactor = 0.8 * ZoomRatio * 1.5
                    -- [FIX] نكتب Position و Size بس لما تتغير بشكل ملحوظ (tolerance 0.5px)
                    local newPX = offsetX * distanceFactor
                    local newPY = offsetY * distanceFactor
                    local currentSize = data.BaseSize * sizePulse
                    local lastPX = data._lastPX or 0
                    local lastPY = data._lastPY or 0
                    local lastSZ = data._lastSZ or 0
                    if math.abs(newPX - lastPX) > 0.5 or math.abs(newPY - lastPY) > 0.5 then
                        data.Instance.Position = UDim2.new(0.5, newPX, 0.5, newPY)
                        data._lastPX = newPX; data._lastPY = newPY

                    if math.abs(currentSize - lastSZ) > 0.5 then
                        data.Instance.Size = UDim2.new(0, currentSize, 0, currentSize)
                        data._lastSZ = currentSize
                    end
                    local baseTrans = 0.983 + (data.Progress * 0.008)
                    local finalTrans = 1 - ((1 - baseTrans) * GlowAlpha)

                    local lastC = data._lastGlowColor

                        or math.abs(lastC.B - CurrentGlowColor.B) > 0.004 then

                        data._lastGlowColor = CurrentGlowColor
                    end
                end
            end

local function FadeUI(element, show)
    local targetAlpha = show and 0 or 1
    local ti = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    if show then element.Visible = true end
    if element:IsA("TextButton") then TweenService:Create(element, ti, {TextTransparency=targetAlpha}):Play() end
    for _, v in pairs(element:GetDescendants()) do
        if v:IsA("UIStroke") then TweenService:Create(v, ti, {Transparency=targetAlpha}):Play()
        elseif v:IsA("ImageLabel") then TweenService:Create(v, ti, {ImageTransparency=show and 0.6 or 1}):Play()
        elseif v:IsA("TextLabel") then TweenService:Create(v, ti, {TextTransparency=targetAlpha}):Play()
        elseif v:IsA("TextBox") then TweenService:Create(v, ti, {TextTransparency=targetAlpha}):Play() end
    end
    if not show then task.delay(0.3, function() if not (_G_settings.UsePrediction and _G_settings.AutoPred) then element.Visible = false end end) end
end

local function CreateSlider(parent, nameKey, min, max, startVal, callback)
    local T = Translations[1] or {}
    local container = Instance.new("Frame", parent)
    container.Size = UDim2.new(1,-20,0,50)
    container.BackgroundColor3 = Color3.fromRGB(0,0,0)
    container.BackgroundTransparency = 0.90
    Instance.new("UICorner", container).CornerRadius = UDim.new(0,4)
    table.insert(AllUIBoxes, container)
    local label = Instance.new("TextLabel", container)
    label.Size = UDim2.new(1,-10,0,25); label.Position = UDim2.new(0,5,0,5)
    label.BackgroundTransparency = 1
    label.Text = (T[nameKey] or nameKey)..": "..math.floor(startVal)
    label.TextColor3 = Color3.fromRGB(255,255,255); label.Font = Enum.Font.Gotham
    SafeTextScale(label,8); label.TextXAlignment = Enum.TextXAlignment.Left
    local sliderBg = Instance.new("Frame", container)
    sliderBg.Size = UDim2.new(1,-20,0,8); sliderBg.Position = UDim2.new(0,10,0,32)
    sliderBg.BackgroundColor3 = Color3.fromRGB(255,255,255); sliderBg.BackgroundTransparency = 0.80
    sliderBg.BorderSizePixel = 0; Instance.new("UICorner", sliderBg).CornerRadius = UDim.new(1,0)
    sliderBg:SetAttribute("IsSliderFill", true)
    table.insert(AllUIBoxes, sliderBg)
    local fill = Instance.new("Frame", sliderBg)
    local perc = (startVal - min) / (max - min)
    fill.Size = UDim2.new(perc,0,1,0)
    fill.BackgroundColor3 = Color3.fromRGB(255,255,255); fill.BackgroundTransparency = 0.20
    fill:SetAttribute("IsSliderFill", true)
    table.insert(AllUIBoxes, fill)
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1,0)
    -- [FIX] نستخدم registry مركزي بدل connection جديدة لكل slider
    _sliderIdCounter = _sliderIdCounter + 1
    local _sid = _sliderIdCounter
    local function updateSlider(input)
        local relX = input.Position.X - sliderBg.AbsolutePosition.X
        local clampX = math.clamp(relX, 0, sliderBg.AbsoluteSize.X)
        local p = clampX / sliderBg.AbsoluteSize.X
        fill.Size = UDim2.new(p,0,1,0)
        local val = min + (max - min) * p
        local currentT = Translations[CurrentLang] or {}
        label.Text = (currentT[nameKey] or nameKey)..": "..math.floor(val)
        callback(val)
    end
    _sliderCallbacks[_sid] = {isDragging = false, updateFn = updateSlider}
    sliderBg.InputBegan:Connect(function(input)
        if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
            _sliderCallbacks[_sid].isDragging = true; updateSlider(input)
        end
    end)
    fill.InputBegan:Connect(function(input)
        if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
            _sliderCallbacks[_sid].isDragging = true; updateSlider(input)
        end
    end)
    -- دالة لتحديث الـ slider بصرياً من برا بدون input
    return function(val)
        local p = math.clamp((val - min) / (max - min), 0, 1)
        fill.Size = UDim2.new(p, 0, 1, 0)
        local currentT = Translations[CurrentLang] or {}
        label.Text = (currentT[nameKey] or nameKey)..": "..math.floor(val)
        callback(val)
    end
end

local _raycastLoaded = false

-- [FIX] دالة مستقلة تثبّت الـ hook مرة واحدة فقط
-- تُستدعى من الزر ومن ApplyRcTargetFromSettings (عند تحميل Gun Mode برمجياً)
-- هذا يحل مشكلة الحاجة لضغطتين: الأولى كانت تثبّت الـ hook وتشغّله،
-- والثانية تطفيه لأن الزر كان يعكس الحالة بعد التثبيت
local function EnsureRaycastHookInstalled()
    if _raycastLoaded then return end
    _raycastLoaded = true
    task.spawn(function()
        local Players = game:GetService("Players")
        local LocalPlayer = Players.LocalPlayer
        local Hooks = {}
        local cachedRoots = {}
        local _rebuildPending = false
        local function rebuildCache()
            -- [OPT] تأجيل الـ rebuild بـ 0.5s لمنع rebuild متكرر عند spawn متعدد
            if _rebuildPending then return end
            _rebuildPending = true
            task.delay(0.5, function()
                _rebuildPending = false
                cachedRoots = {}
                for _, player in ipairs(Players:GetPlayers()) do
                    if player ~= LocalPlayer and player.Character then
                        local root = player.Character:FindFirstChild("HumanoidRootPart")
                        local head = player.Character:FindFirstChild("Head")
                        local hum = player.Character:FindFirstChildOfClass("Humanoid")
                        if root and head and hum then
                            table.insert(cachedRoots, {root=root, head=head, hum=hum})
                        end
                    end
                end
            end)
        end
        Players.PlayerAdded:Connect(rebuildCache)
        Players.PlayerRemoving:Connect(rebuildCache)
        Players.PlayerAdded:Connect(function(p) p.CharacterAdded:Connect(rebuildCache) end)
        for _, p in ipairs(Players:GetPlayers()) do p.CharacterAdded:Connect(rebuildCache) end
        rebuildCache()
        -- Wall Check: نحسب visibility كل 0.15s في loop منفصل بدل داخل الـ hook
        local _rcWallParams = RaycastParams.new()
        _rcWallParams.FilterType = Enum.RaycastFilterType.Exclude
        local _rcVisibleSet = {} -- cache: entry.root -> true/false

        task.spawn(function()
            while true do
                task.wait(0.25)
                -- [OPT] انتظر أطول لما الـ hook مطفي كاملاً
                if not _rcTarget.enabled then
                    _rcVisibleSet = {}
                    task.wait(0.5)
                    continue
                end
                if not _rcTarget.wallCheck then
                    _rcVisibleSet = {}
                    continue
                end
                local myChar = lp.Character
                local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
                if not myRoot then continue end
                _rcWallParams.FilterDescendantsInstances = {myChar}
                local origin = myRoot.Position
                local newVisible = {}
                for _, entry in ipairs(cachedRoots) do
                    if entry.root and entry.root.Parent and entry.head then
                        local dir = entry.head.Position - origin
                        local result = workspace:Raycast(origin, dir, _rcWallParams)
                        if result then
                            local hitModel = result.Instance:FindFirstAncestorOfClass("Model")
                            newVisible[entry.root] = (hitModel == entry.root.Parent)
                        else
                            newVisible[entry.root] = true -- مفتوح، ما في شي يعترض
                        end
                    end
                end
                _rcVisibleSet = newVisible
            end
        end)

        -- [OPT] بدل بناء جدول candidates ثم إعادة loop عليه →
        -- نمشي على الـ cachedRoots مرة وحدة ونختار الأفضل مباشرة
        local function getClosest(origin)
            local bestEntry = nil
            local bestVal   = math.huge
            local bestDist  = math.huge
            local distLimit = _rcTarget.dist
            local lowHP     = _rcTarget.lowHealth
            local wallOn    = _rcTarget.wallCheck

            local distLimitSq = distLimit * distLimit
            for _, entry in ipairs(cachedRoots) do
                if entry.root and entry.root.Parent and entry.hum.Health > 0 then
                    -- [OPT] مقارنة تربيعية أسرع من .Magnitude
                    local rp = entry.root.Position
                    local dx = rp.X - origin.X
                    local dy = rp.Y - origin.Y
                    local dz = rp.Z - origin.Z
                    local distSq = dx*dx + dy*dy + dz*dz
                    if distSq > distLimitSq then continue end
                    local dist = math.sqrt(distSq)

                    -- [[ زر 1: Wall Check - من الـ cache، ما في Raycast هنا ]]
                    if wallOn and _rcVisibleSet[entry.root] == false then continue end

                    -- [[ زر 2: HP أو Distance ]]
                    local val = lowHP and entry.hum.Health or dist
                    if val < bestVal or (val == bestVal and dist < bestDist) then
                        bestVal   = val
                        bestDist  = dist
                        bestEntry = entry
                    end
                end
            end

            if not bestEntry then return nil end
            local pos = bestEntry.head.Position
            if _rcTarget.pred > 0 then
                local vel = bestEntry.root.AssemblyLinearVelocity
                if vel.Magnitude > 130 then vel = vel.Unit * 130 end
                pos = pos + vel * _rcTarget.pred
            end
            return pos
        end
        -- [OPT] Cache method check - getnamecallmethod() سريع لكن lowercase() مو مجاني
        -- نتجنب string ops إضافية لما الـ hook مطفي
        Hooks.Raycast = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
            -- [OPT] Early exit أسرع ما يمكن قبل أي عملية
            if not _rcTarget.enabled then
                return Hooks.Raycast(self, ...)
            end
            if self.Name ~= "Workspace" then
                return Hooks.Raycast(self, ...)
            end
            local _method = getnamecallmethod()
            if _method ~= "raycast" and _method ~= "Raycast" then
                return Hooks.Raycast(self, ...)
            end
            local args = {...}
            local origin = args[1]
            if typeof(origin) == "Vector3" then
                local closest = getClosest(origin)
                if closest then
                    args[2] = (closest - origin).Unit * math.clamp(_rcTarget.speed, 10, 40)
                    return Hooks.Raycast(self, table.unpack(args))
                end
            end
            return Hooks.Raycast(self, ...)
        end))
    end)
end

local ActiveEditorSlider = false
local function CreateEditorSlider(parentUI, name, yPos, min, max, startVal, callback)
    local label = Instance.new("TextLabel", parentUI)
    label.Size = UDim2.new(1,-20,0,20); label.Position = UDim2.new(0,10,0,yPos); label.BackgroundTransparency = 1; label.Text = name..": "..math.floor(startVal); label.TextColor3 = Color3.new(1,1,1); label.Font = Enum.Font.Gotham; label.TextSize = 12; label.TextXAlignment = Enum.TextXAlignment.Left
    local container = Instance.new("Frame", parentUI)
    container.Size = UDim2.new(1,-20,0,20); container.Position = UDim2.new(0,10,0,yPos+20); container.BackgroundColor3 = Color3.fromRGB(255,255,255); container.BackgroundTransparency = 0.90
    Instance.new("UICorner", container).CornerRadius = UDim.new(0,4)
    local fill = Instance.new("Frame", container)
    local perc = (startVal-min)/(max-min)
    fill.Size = UDim2.new(perc,0,1,0); fill.BackgroundColor3 = Color3.fromRGB(255,0,0);
    local isDragging = false
    container.InputBegan:Connect(function(input) if not ActiveEditorSlider and (input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch) then isDragging=true; ActiveEditorSlider=true end end)
    fill.InputBegan:Connect(function(input) if not ActiveEditorSlider and (input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch) then isDragging=true; ActiveEditorSlider=true end end)
    UserInputService.InputEnded:Connect(function(input) if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then if isDragging then isDragging=false; ActiveEditorSlider=false end end end)
    -- [OPT] parentUI.Visible هنا كافي - ما نحتاج guard إضافي لأن Editor Gui مستقل
    UserInputService.InputChanged:Connect(function(input)
        if isDragging and (input.UserInputType==Enum.UserInputType.MouseMovement or input.UserInputType==Enum.UserInputType.Touch) and parentUI.Visible then
            local relativeX = input.Position.X - container.AbsolutePosition.X; local clampedX = math.clamp(relativeX,0,container.AbsoluteSize.X)
            local p = clampedX/container.AbsoluteSize.X; fill.Size = UDim2.new(p,0,1,0)
            local val = min + (max-min)*p; label.Text = name..": "..math.floor(val); callback(val)
        end
    end)
    return function(newVal) local p = (newVal-min)/(max-min); fill.Size = UDim2.new(p,0,1,0); label.Text = name..": "..math.floor(newVal) end
end

-- [OPT] نحفظ الـ tweens النشطة لكل page عشان نلغيها قبل ما ننشئ جديدة
-- بدون هاد كل ضغطة على تاب تراكم tweens في الذاكرة مع الوقت
local _activePageTweens = {}

local function FadePage(page, show)
    local ti = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)

    -- الغِ كل الـ tweens القديمة لهاي الصفحة قبل ما تبدأ جديدة
    if _activePageTweens[page] then
        for _, tw in ipairs(_activePageTweens[page]) do
            pcall(function() tw:Cancel() end)
        end
    end
    _activePageTweens[page] = {}

    local tweens = _activePageTweens[page]

    if show then
        page.Visible = true
        for _, v in ipairs(page:GetDescendants()) do
            local tw
            if v:IsA("TextLabel") then
                v.TextTransparency = 1
                tw = TweenService:Create(v, ti, {TextTransparency = 0})
            elseif v:IsA("TextButton") then
                if v:GetAttribute("ColorName") then
                    -- skip color square buttons, keep their original color
                else
                    v.TextTransparency = 1
                    v.BackgroundTransparency = 1
                    if v.BackgroundColor3 == Color3.fromRGB(255,255,255) or v.BackgroundColor3 == Color3.new(1,1,1) then
                        v.BackgroundColor3 = CurrentBoxBgColor
                    end
                    tw = TweenService:Create(v, ti, {TextTransparency = 0, BackgroundTransparency = 0.90, BackgroundColor3 = CurrentBoxBgColor})
                end
            elseif v:IsA("TextBox") then
                v.TextTransparency = 1
                v.BackgroundTransparency = 1
                if v.BackgroundColor3 == Color3.fromRGB(255,255,255) or v.BackgroundColor3 == Color3.new(1,1,1) then
                    v.BackgroundColor3 = CurrentBoxBgColor
                end
                tw = TweenService:Create(v, ti, {TextTransparency = 0, BackgroundTransparency = 0.90, BackgroundColor3 = CurrentBoxBgColor})
            elseif v:IsA("Frame") then
                if v:GetAttribute("IsJoystick") or v:GetAttribute("IsSliderFill") then
                    -- slider/joystick: fade in بدون تغيير اللون
                    local origTrans = v:GetAttribute("OrigTransparency")
                    if origTrans == nil then origTrans = v.BackgroundTransparency end
                    v:SetAttribute("OrigTransparency", origTrans)
                    v.BackgroundTransparency = 1
                    tw = TweenService:Create(v, ti, {BackgroundTransparency = origTrans})
                elseif v.BackgroundColor3 == Color3.fromRGB(255,255,255) or v.BackgroundColor3 == Color3.new(1,1,1) then
                    v.BackgroundTransparency = 1
                    v.BackgroundColor3 = CurrentBoxBgColor
                    tw = TweenService:Create(v, ti, {BackgroundTransparency = 0.90, BackgroundColor3 = CurrentBoxBgColor})
                else
                    v.BackgroundTransparency = 1
                    tw = TweenService:Create(v, ti, {BackgroundTransparency = 0.90})
                end
            end
            if tw then tw:Play(); table.insert(tweens, tw) end
        end
    else
        -- [OPT] الصفحة مخفية أصلاً - مافي حاجة نعمل tweens عليها
        if not page.Visible then _activePageTweens[page] = nil; return end
        for _, v in ipairs(page:GetDescendants()) do
            local tw
            if v:IsA("TextLabel") then
                tw = TweenService:Create(v, ti, {TextTransparency = 1})
            elseif v:IsA("TextButton") then
                tw = TweenService:Create(v, ti, {TextTransparency = 1, BackgroundTransparency = 1})
            elseif v:IsA("TextBox") then
                tw = TweenService:Create(v, ti, {TextTransparency = 1, BackgroundTransparency = 1})
            elseif v:IsA("Frame") then
                if v:GetAttribute("IsJoystick") or v:GetAttribute("IsSliderFill") then
                    -- slider/joystick: fade out بدون تغيير اللون، ونحفظ الـ transparency الأصلية
                    local origTrans = v:GetAttribute("OrigTransparency")
                    if origTrans == nil then origTrans = v.BackgroundTransparency end
                    v:SetAttribute("OrigTransparency", origTrans)
                    tw = TweenService:Create(v, ti, {BackgroundTransparency = 1})
                else
                    tw = TweenService:Create(v, ti, {BackgroundTransparency = 1})
                end
            end
            if tw then tw:Play(); table.insert(tweens, tw) end
        end
        task.delay(0.3, function()
            page.Visible = false
            -- بعد ما تخلص الـ tweens نمسحها من الذاكرة
            _activePageTweens[page] = nil
        end)
    end
end

local function SwitchTab(activeKey)
    if currentActiveTab == activeKey then return end
    currentActiveTab = activeKey

    for key, page in pairs(TabPageMap) do
        local tabBtn = UI_Elements[key]
        local uiScale = tabBtn and tabBtn:FindFirstChildOfClass("UIScale")

        if key == activeKey then
            FadePage(page, true)
            if tabBtn then
                tabBtn:SetAttribute("IsActive", true)
                TweenService:Create(tabBtn, TweenInfo.new(0.2, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {BackgroundTransparency = 1}):Play()
                task.delay(0.18, function()
                    if uiScale then TweenService:Create(uiScale, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1.10}):Play() end
                end)
            end
        else
            FadePage(page, false)
            if tabBtn then
                tabBtn:SetAttribute("IsActive", false)
                if uiScale then TweenService:Create(uiScale, TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {Scale = 1.0}):Play() end
                TweenService:Create(tabBtn, TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {BackgroundTransparency = 0.90}):Play()
            end
        end
    end
end

-- [OPT] Cache RaycastParams مرة واحدة خارج الـ RenderStepped بدل ما تُنشأ كل فريم
local _aimRayParams = RaycastParams.new()

local _fovColor = Color3.fromRGB(255,0,0)
-- [OPT] Cache آخر قيم FOVCircle - نكتب الـ property بس لما تتغير القيمة فعلاً
-- بدل ما نكتب نفس القيمة 60+ مرة/ثانية (كل كتابة property = overhead على الـ render thread)
local _lastFovRadius  = nil
local _lastFovOffsetY = nil
local _lastFovVpX     = nil
local _lastFovVpY     = nil
local _aimCamCenterCache = nil  -- cache لمركز الشاشة، يتحدث بس لما ViewportSize يتغير

-- [OPT] دالة مشتركة لحساب الـ prediction وتطبيق الـ aim على الكاميرا
local function applyAimToCam(cam, myRoot, bestPart, deltaTime)
    local aimPos = bestPart.Position
    if _G_settings.UsePrediction then
        local currentPrediction = _G_settings.Prediction
        if _G_settings.AutoPred then
            local dist = (myRoot.Position - bestPart.Position).Magnitude
            currentPrediction = _G_settings.AutoPredBase
            if dist <= _G_settings.DistThreshold then
                currentPrediction = currentPrediction + (dist * _G_settings.CloseMult)
            else
                currentPrediction = currentPrediction + (_G_settings.DistThreshold * _G_settings.CloseMult) + ((dist - _G_settings.DistThreshold) * _G_settings.FarMult)
            end
            if UI_Inputs.Pred then UI_Inputs.Pred.Text = string.format("%.3f", currentPrediction) end
        end
        local vel = bestPart.AssemblyLinearVelocity
        if vel.Magnitude > 130 then vel = vel.Unit * 130 end
        aimPos = aimPos + Vector3.new(vel.X * currentPrediction, vel.Y * currentPrediction * _G_settings.VerticalStrength, vel.Z * currentPrediction)
    end
    local targetLook = CFrame.lookAt(cam.CFrame.Position, aimPos)
    local offsetAngle = math.rad((_G_settings.CrosshairOffsetY / cam.ViewportSize.Y) * cam.FieldOfView)
    targetLook = targetLook * CFrame.Angles(offsetAngle, 0, 0)
    if targetLook.X == targetLook.X then -- NaN check
        if not followingPath then
            local smoothVal = 1
            if _G_settings.Smoothness and _G_settings.Smoothness > 0 then
                smoothVal = math.clamp(1 / (_G_settings.Smoothness * deltaTime * 60), 0.01, 1)
            end
            cam.CFrame = cam.CFrame:Lerp(targetLook, smoothVal)
        end
    end
end

RunService.RenderStepped:Connect(function(deltaTime)
    local cam = workspace.CurrentCamera
    if not cam then return end

    -- FOVCircle + Aimbot: لا داعي لأي عمل لما الـ Aimbot مطفي كاملاً
    if not _G_settings.Enabled then
        -- نخفي الـ FOVCircle بس مرة وحدة لما يتطفى، مش كل فريم
        if FOVCircle and FOVCircle.Visible then FOVCircle.Visible = false end
        return
    end

    -- FOVCircle update - [OPT] نكتب الـ properties بس لما تتغير القيمة فعلاً
    if FOVCircle then
        local vpX, vpY   = cam.ViewportSize.X, cam.ViewportSize.Y
        local offsetY     = _G_settings.CrosshairOffsetY
        local newRadius   = _G_settings.FOV
        -- Position: نحدّث بس لما ViewportSize أو CrosshairOffsetY يتغيروا
        if vpX ~= _lastFovVpX or vpY ~= _lastFovVpY or offsetY ~= _lastFovOffsetY then
            FOVCircle.Position = Vector2.new(vpX / 2, (vpY / 2) + offsetY)
            _lastFovVpX     = vpX
            _lastFovVpY     = vpY
            _lastFovOffsetY = offsetY
        end
        -- Radius: نحدّث بس لما FOV يتغير
        if newRadius ~= _lastFovRadius then
            FOVCircle.Radius = newRadius
            _lastFovRadius   = newRadius
        end
        -- Visible: بس لما HideFOV يتغير
        local shouldShow = not _G_settings.HideFOV
        if FOVCircle.Visible ~= shouldShow then FOVCircle.Visible = shouldShow end
    end

    if _G_settings.RightClickAim and not isAimKeyDown then return end

    local myChar = lp.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    local bestTarget = nil
    local bestPart   = nil
    local bestVal    = math.huge

    -- ============================================================
    -- وضع FOV مطفي → Distance Mode
    -- يختار أقرب عدو بالمسافة بغض النظر عن اتجاه الكاميرا
    -- يشمل الأعداء خلف اللاعب
    -- ============================================================
    if _G_settings.HideFOV then
        local myPos = myRoot.Position
        for v, ent in pairs(LiveTargets) do
            if _G_settings.IgnoredPlayers[ent.name] then continue end
            if v and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
                -- [FIX] استخدام الـ cache من LiveTargets بدل FindFirstChild كل فريم
                local targetPart = ent[_G_settings.TargetPart] or v:FindFirstChild(_G_settings.TargetPart)
                if targetPart then
                    local worldDist = (myPos - targetPart.Position).Magnitude
                    if worldDist <= _G_settings.MaxDist then
                        -- إذا LowHealthFocus مفعّل → يفضّل الأقل صحة، وإلا يفضّل الأقرب
                        local checkVal = _G_settings.LowHealthFocus and v.Humanoid.Health or worldDist
                        if checkVal < bestVal then
                            local isVisible = true
                            if _G_settings.WallCheck then
                                _aimRayParams.FilterDescendantsInstances = {myChar, v}
                                local ray = workspace:Raycast(myPos, (targetPart.Position - myPos).Unit * worldDist, _aimRayParams)
                                if ray then isVisible = false end
                            end
                            if isVisible then
                                bestVal = checkVal
                                bestTarget = v
                                bestPart = targetPart
                            end
                        end
                    end
                end
            end
        end

    -- ============================================================
    -- وضع FOV مفعّل → FOV Screen Mode (النظام الأصلي)
    -- يختار أقرب عدو لمركز الشاشة ضمن دائرة الـ FOV
    -- ============================================================
    else
        -- [OPT] camCenter: نحسبه بس لما ViewportSize يتغير، بدل ما ننشئ Vector2.new جديد كل فريم
        -- ViewportSize يتغير نادراً جداً (فقط لما اللاعب يغير حجم النافذة)
        local vpSize = cam.ViewportSize
        if not _aimCamCenterCache
            or _aimCamCenterCache._vpX ~= vpSize.X
            or _aimCamCenterCache._vpY ~= vpSize.Y
            or _aimCamCenterCache._offsetY ~= _G_settings.CrosshairOffsetY then
            _aimCamCenterCache = {
                vec     = Vector2.new(vpSize.X/2, (vpSize.Y/2) + _G_settings.CrosshairOffsetY),
                _vpX    = vpSize.X,
                _vpY    = vpSize.Y,
                _offsetY = _G_settings.CrosshairOffsetY
            }
        end
        local camCenter = _aimCamCenterCache.vec
        for v, ent in pairs(LiveTargets) do
            if _G_settings.IgnoredPlayers[ent.name] then continue end
            if v and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
                -- [FIX] استخدام الـ cache من LiveTargets بدل FindFirstChild كل فريم
                local targetPart = ent[_G_settings.TargetPart] or v:FindFirstChild(_G_settings.TargetPart)
                if targetPart then
                    local worldDist = (myRoot.Position - targetPart.Position).Magnitude
                    if worldDist <= _G_settings.MaxDist then
                        local screenPos, onScreen = cam:WorldToViewportPoint(targetPart.Position)
                        if onScreen then
                            local mouseDist = (Vector2.new(screenPos.X, screenPos.Y) - camCenter).Magnitude
                            if mouseDist <= _G_settings.FOV then
                                local checkVal = _G_settings.LowHealthFocus and v.Humanoid.Health or mouseDist
                                if checkVal < bestVal then
                                    local isVisible = true
                                    if _G_settings.WallCheck then
                                        _aimRayParams.FilterDescendantsInstances = {myChar, v}
                                        local ray = workspace:Raycast(cam.CFrame.Position, (targetPart.Position - cam.CFrame.Position).Unit * worldDist, _aimRayParams)
                                        if ray then isVisible = false end
                                    end
                                    if isVisible then
                                        bestVal = checkVal
                                        bestTarget = v
                                        bestPart = targetPart
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    -- تطبيق الـ aim على أفضل هدف تم إيجاده
    if bestTarget and bestPart then
        applyAimToCam(cam, myRoot, bestPart, deltaTime)
    end
end)

    elseif _savedColor == "LightBlue" then ApplyColorTheme(Color3.fromRGB(0, 200, 255), LightBlueBtn_c)
    elseif _savedColor == "NormalRed" then ApplyColorTheme(Color3.fromRGB(255, 0, 0), NormalRedBtn_c)
    elseif _savedColor == "DarkRed" then ApplyColorTheme(Color3.fromRGB(130, 0, 0), DarkRedBtn_c)
    elseif _savedColor == "Pink" then ApplyColorTheme(Color3.fromRGB(255, 105, 180), PinkBtn_c)
    elseif _savedColor == "DarkPink" then ApplyColorTheme(Color3.fromRGB(180, 0, 100), DarkPinkBtn_c)
    elseif _savedColor == "LightPink" then ApplyColorTheme(Color3.fromRGB(255, 182, 213), LightPinkBtn_c)
    else ApplyColorTheme(Color3.fromRGB(255, 0, 0), NormalRedBtn_c)
    end
    -- تحديث حالة أزرار Raycast بعد ما كل الـ UI اتبنى
    UpdateRaycastSquareBtns()
end
