local function openDiscord()

    local code = "z3Pt4saTx"

    local fullUrl = "https://discord.gg/" .. code



    local copyFunc = setclipboard or toclipboard or (syn and syn.write_clipboard)

    if copyFunc then pcall(copyFunc, fullUrl) end



    local reqFunc = request or (syn and syn.request) or (http and http.request) or (fluxus and fluxus.request)

    if reqFunc then

        pcall(reqFunc, {

            Url = "http://127.0.0.1:6463/rpc?v=1",

            Method = "POST",

            Headers = {

                ["Content-Type"] = "application/json",

                ["Origin"] = "https://discord.com",

            },

            Body = HttpService:JSONEncode({

                cmd = "INVITE_BROWSER",

                args = { code = code },

                nonce = HttpService:GenerateGUID(false),

            }),

            Timeout = 2,

        })

    end

end

local lastMousePos = nil



local function drawLineSegment(p1, p2)

    local dist = (p2 - p1).Magnitude

    if dist < 2 then return end



    local angle = math.deg(math.atan2(p2.Y - p1.Y, p2.X - p1.X))

    local center = (p1 + p2) / 2



    local line = Instance.new("Frame")

    line.Size = UDim2.new(0, dist, 0, 3)

    line.Position = UDim2.new(0, center.X, 0, center.Y)

    line.AnchorPoint = Vector2.new(0.5, 0.5)

    line.Rotation = angle

    line.BackgroundColor3 = Color3.fromRGB(190, 60, 255)

    line.BorderSizePixel = 0

    line.ZIndex = 3

    line.Parent = trailFrame



    Instance.new("UICorner", line).CornerRadius = UDim.new(1, 0)



    local stroke = Instance.new("UIStroke")

    stroke.Color = Color3.fromRGB(230, 120, 255)

    stroke.Thickness = 2.5

    stroke.Transparency = 0.2

    stroke.Parent = line



    local tw = TweenService:Create(line, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {

        BackgroundTransparency = 1,

        Size = UDim2.new(0, dist, 0, 0)

    })

    TweenService:Create(stroke, TweenInfo.new(0.5), {Transparency = 1}):Play()



    tw:Play()

    tw.Completed:Connect(function() line:Destroy() end)

end

local MeleeModule = nil
pcall(function()
    local loaded = (typeof(getloadedmodules) == "function" and getloadedmodules()) or {}
    for _, mod in pairs(loaded) do 
        if mod.Name == "MeleeWeaponClient" then MeleeModule = require(mod) break end 
    end
    if not MeleeModule then MeleeModule = require(ReplicatedStorage.Client.Source.Melee.MeleeWeaponClient) end
end)

local function getMeleeObject(tool)
    if not tool or not tool:IsA("Tool") then return nil end
    local tName = string.lower(tool.Name)
    if string.find(tName, "chaos") or string.find(tName, "blade") or string.find(tName, "dual") then
        local success, obj = pcall(function() return MeleeModule and (MeleeModule.getObj(tool) or MeleeModule:GetWeaponObject(tool)) end)
        if success and obj then return obj end
    end
    if MeleeModule and type(MeleeModule.getObj) == "function" then 
        local success, obj = pcall(MeleeModule.getObj, tool)
        if success then return obj end
    end
    return nil
end

local function startAimbotLogic()
    local Camera = workspace.CurrentCamera
    fovCircleInstance = Drawing.new("Circle")
    fovCircleInstance.Color = Color3.fromRGB(255, 255, 255)
    fovCircleInstance.Thickness = 1
    fovCircleInstance.NumSides = 64
    fovCircleInstance.Filled = false
    fovCircleInstance.Visible = false

    regConn(RunService.RenderStepped:Connect(function()
        if fovCircleInstance then
            fovCircleInstance.Visible = getgenv().AimFovEnabled and getgenv().AimbotEnabled
            fovCircleInstance.Radius = getgenv().AimFovRadius or 120
            fovCircleInstance.Position = UserInputService:GetMouseLocation()
        end
        if not getgenv().AimbotEnabled then return end
        if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end
        local closestPlayer = nil
        local shortestDistance = math.huge
        local myChar = localPlayer.Character
        local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if not myHrp then return end
        local mouseLocation = UserInputService:GetMouseLocation()
        for _, player in pairs(Players:GetPlayers()) do
            if player ~= localPlayer and player.Character then
                local boneName = getgenv().AimPart or "HumanoidRootPart"
                local targetBone = player.Character:FindFirstChild(boneName)
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if targetBone and hum and hum.Health > 0 then
                    if getgenv().DownedParryEnabled and player.Character:GetAttribute("Downed") then continue end
                    if getgenv().AimWallCheck then
                        local raycastParams = RaycastParams.new()
                        raycastParams.FilterType = Enum.RaycastFilterType.Exclude
                        raycastParams.FilterDescendantsInstances = { myChar, player.Character }
                        local rayResult = workspace:Raycast(Camera.CFrame.Position, targetBone.Position - Camera.CFrame.Position, raycastParams)
                        if rayResult then continue end
                    end
                    local screenPos, onScreen = Camera:WorldToViewportPoint(targetBone.Position)
                    if getgenv().AimFovEnabled then
                        local distFromMouse = (Vector2.new(screenPos.X, screenPos.Y) - mouseLocation).Magnitude
                        if distFromMouse > (getgenv().AimFovRadius or 120) then continue end
                    end
                    if getgenv().AimTargetMode == "Cursor" then
                        local distFromMouse = (Vector2.new(screenPos.X, screenPos.Y) - mouseLocation).Magnitude
                        if distFromMouse < shortestDistance then shortestDistance = distFromMouse closestPlayer = player end
                    else
                        local distance = (targetBone.Position - myHrp.Position).Magnitude
                        if distance < shortestDistance and distance <= ((getgenv().ParryRange or 16.5) * 3) then shortestDistance = distance closestPlayer = player end
                    end
                end
            end
        end
        if closestPlayer and closestPlayer.Character then
            local boneName = getgenv().AimPart or "HumanoidRootPart"
            local targetBone = closestPlayer.Character:FindFirstChild(boneName)
            if targetBone then
                local finalTargetPos = targetBone.Position
                if getgenv().AimPrediction and getgenv().AimPrediction > 0 then
                    finalTargetPos = finalTargetPos + (targetBone.Velocity * (getgenv().AimPrediction / 10))
                end
                local screenPos, onScreen = Camera:WorldToViewportPoint(finalTargetPos)
                if onScreen then
                    local deltaX = screenPos.X - mouseLocation.X
                    local deltaY = screenPos.Y - mouseLocation.Y
                    local smoothness = math.max(1, getgenv().AimSmoothness or 5)
                    if type(mousemoverel) == "function" then mousemoverel(deltaX / smoothness, deltaY / smoothness)
                    else
                        local targetLook = CFrame.lookAt(Camera.CFrame.Position, finalTargetPos)
                        Camera.CFrame = Camera.CFrame:Lerp(targetLook, 1 / smoothness)
                    end
                end
            end
        end
    end))
end

        raycastParams.FilterDescendantsInstances = { localPlayer.Character, targetPart.Parent }

    local function calculateThreatIndex(attackerChar, attackerPlayer)
        local myHRP, myChar = getLocalComponents()
        if not myHRP or not attackerChar then return 0, "No Base Components" end
        
        local attHRP = attackerChar:FindFirstChild("HumanoidRootPart")
        if not attHRP then return 0, "No Attacker HRP" end

            return 0, "Target Occluded By Geometry" 
        end

        local parts = workspace:GetPartsInPart(parryHitbox)
        local isInside = false

        for _, part in ipairs(parts) do
            if part:IsDescendantOf(attackerChar) then
                isInside = true
                break
            end
        end

            return 0, "Target Outside Hitbox"
        end

        return 1.0, "Success"

    local function findMobileParryButton()
        -- Only target the game's mobile combat button.
        -- Never scan the hub UI itself, otherwise "Mobile Auto Parry"
        -- can be mistaken for the game's Parry button and activate the hub.
        local playerGui = localPlayer:FindFirstChildOfClass("PlayerGui")
        if not playerGui then return nil end

        local candidates = {}
        for _, obj in ipairs(playerGui:GetDescendants()) do
            if obj:IsA("GuiButton") and obj.Visible and obj.Active then
                local ancestor = obj
                local insideHub = false

                while ancestor do
                    if ancestor.Name == "TrunkistHubUI"
                        or ancestor.Name == "TrunkistWelcomeUI"
                        or ancestor.Name == "TrunkistHitmarkerGUI"
                        or ancestor.Name == "TrunkistWatermark" then
                        insideHub = true
                        break
                    end
                    ancestor = ancestor.Parent
                end

                if not insideHub then
                    local name = string.lower(obj.Name or "")
                    local label = string.lower(obj:IsA("TextButton") and obj.Text or "")
                    if string.find(name, "parry") or string.find(name, "block")
                        or string.find(name, "deflect") or string.find(label, "parry")
                        or string.find(label, "block") or string.find(label, "deflect") then
                        table.insert(candidates, obj)
                    end
                end
            end
        end

        -- Prefer an actual mobile combat control over generic matching buttons.
        for _, obj in ipairs(candidates) do
            local name = string.lower(obj.Name or "")
            local label = string.lower(obj:IsA("TextButton") and obj.Text or "")
            if string.find(name, "parry") or string.find(label, "parry")
                or string.find(name, "deflect") or string.find(label, "deflect") then
                return obj
            end
        end

        return candidates[1]
    end

    local function triggerParry(otherPlayer)
        if not getgenv().MasterParryEnabled then return end
        local now = os.clock()
        local lastTime = lastParryTimes[otherPlayer] or 0
        if now - lastTime < PARRY_COOLDOWN then return end
        
        local otherChar = otherPlayer.Character or playerCharacters:FindFirstChild(otherPlayer.Name)
        if not otherChar then return end
        
        local threatScore, logReason = calculateThreatIndex(otherChar, otherPlayer)
        if threatScore < 0.45 then return end
        if math.random(1, 100) > getgenv().ParryChance then return end
        
        local char = localPlayer.Character 
        if not char then return end
        
        local currentTool = char:FindFirstChildOfClass("Tool")
        local meleeObj = getMeleeObject(currentTool)
        
        if not meleeObj and getgenv().AutoEquipParry then
            equipTool()
            currentTool = char:FindFirstChildOfClass("Tool")
            meleeObj = getMeleeObject(currentTool)
        end
        
        lastParryTimes[otherPlayer] = now
        parryCount = parryCount + 1
        pcall(function() 
            StatusLabel.Text = string.format("PING SYNC ACTIVE | PARRIES: %d | THREAT: %.2f", parryCount, threatScore) 
        end)

        if getgenv().HitmarkerEnabled then
            task.spawn(function()
                local sound = Instance.new("Sound") 
                sound.SoundId = "rbxassetid://4817809188" 
                sound.Volume = 2 
                sound.Parent = SoundService 
                sound:Play() 
                Debris:AddItem(sound, 1)
            end)
        end

        startStareAtTarget(otherChar)
        silentBackstabStealth(otherChar)

        if getgenv().ParryDelay and getgenv().ParryDelay > 0 then 
            task.wait(getgenv().ParryDelay / 1000) 
        end
        
        if not meleeObj then 
            pcall(safePressF)
        else 
            pcall(safePressF) 
            pcall(safeAttack, otherChar) 
        end

        stopStareAtTarget()
    end

    local function isAttack(text, keywords)
        local low = string.lower(text)
        for _, word in ipairs(ignoreKeywords) do 
            if string.find(low, word) then return false end 
        end
        for _, word in ipairs(keywords) do 
            if string.find(low, word) then return true end 
        end
        if tonumber(low) and tonumber(low) >= 1 and tonumber(low) <= 5 then return true end
        return false
    end

    local function trackPlayer(otherPlayer)
        cleanPlayerTrack(otherPlayer)
        if otherPlayer == localPlayer then return end
        local currentConns = {}
        playerTracks[otherPlayer] = currentConns

        local function watchCharacter(charFolder)
            local humanoid = charFolder:WaitForChild("Humanoid", 5)
            if humanoid then
                local animator = humanoid:WaitForChild("Animator", 5)
                if animator then
                    local cAnim = animator.AnimationPlayed:Connect(function(track)
                        pcall(function()
                            local id = tostring(track.Animation and track.Animation.AnimationId or "")
                            local name = tostring(track.Name or "")
                            if getgenv().AnimationLogger then 
                                print(string.format("[TRUNKIST-ANIM] Player: %s | Name: %s | ID: %s", otherPlayer.Name, name, id)) 
                            end
                            if string.find(id, "14234031671") or string.find(id, "10467410058") or string.find(string.lower(name), "kick") then return end
                            local lowId, lowName = string.lower(id), string.lower(name)
                            if isAttack(lowId, swingKeywords) or isAttack(lowName, swingKeywords) then 
                                task.spawn(function() 
                                    if checkIsRealSwing(track) then 
                                        triggerParry(otherPlayer) 
                                    end 
                                end)
                            end
                        end)
                    end)
                    table.insert(currentConns, cAnim) 
                    table.insert(scriptConnections, cAnim)
                end
            end

            local function checkAsset(desc)
                pcall(function() 
                    if desc:IsA("Sound") and isAttack(desc.Name, soundKeywords) then 
                        triggerParry(otherPlayer) 
                    end 
                end)
            end

            for _, desc in pairs(charFolder:GetDescendants()) do checkAsset(desc) end
            local cDesc = charFolder.DescendantAdded:Connect(checkAsset)
            table.insert(currentConns, cDesc) 
            table.insert(scriptConnections, cDesc)
        end

        if otherPlayer.Character then watchCharacter(otherPlayer.Character) end
        local cChar = otherPlayer.CharacterAdded:Connect(watchCharacter)
        table.insert(currentConns, cChar) 
        table.insert(scriptConnections, cChar)
    end

    for _, op in pairs(Players:GetPlayers()) do trackPlayer(op) end
    regConn(Players.PlayerAdded:Connect(trackPlayer))
    regConn(Players.PlayerRemoving:Connect(function(op) 
        cleanPlayerTrack(op) 
        lastParryTimes[op] = nil 
    end))
end

--[16. SNAPLINES (TRACERS) LOGIC]--
local function startTracersLogic()
    local Camera = workspace.CurrentCamera
    regConn(RunService.RenderStepped:Connect(function()
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                local hrp = p.Character.HumanoidRootPart
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 and getgenv().TracersEnabled then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                    if onScreen then
                        if not activeTracersLines[p] then
                            local line = Drawing.new("Line") line.Thickness = 1 line.Color = Color3.fromRGB(255, 255, 255)
                            activeTracersLines[p] = line
                        end
                        local line = activeTracersLines[p]
                        line.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                        line.To = Vector2.new(screenPos.X, screenPos.Y)
                        line.Visible = true
                    else if activeTracersLines[p] then activeTracersLines[p].Visible = false end end
                else if activeTracersLines[p] then activeTracersLines[p].Visible = false end end
            else if activeTracersLines[p] then activeTracersLines[p].Visible = false end end
        end
    end))
    regConn(Players.PlayerRemoving:Connect(function(p)
        if activeTracersLines[p] then activeTracersLines[p]:Remove() activeTracersLines[p] = nil end
    end))
end

--[17. ИСПРАВЛЕННЫЙ АНТИ-ПАРРИ]--
local function startFixedAntiParry()
    if not UserInputService then return end
    local isAttacking = false
    local attackEndTime = 0

    local function unequipWeapon()
        local char = localPlayer.Character if not char then return end
        local tool = char:FindFirstChildOfClass("Tool") if not tool then return end
        if tool:FindFirstChild("Unequip") and type(tool.Unequip.Invoke) == "function" then pcall(tool.Unequip.Invoke, tool.Unequip) end
        if type(keypress) == "function" and type(keyrelease) == "function" then
            keypress(0x30) task.wait(0.02) keyrelease(0x30)
        else
            pcall(function()
                local vim = game:GetService("VirtualInputManager")
                vim:SendKeyEvent(true, Enum.KeyCode.Zero, false, game) task.wait(0.02) vim:SendKeyEvent(false, Enum.KeyCode.Zero, false, game)
            end)
        end
        task.delay(0.1, function()
            if type(keypress) == "function" and type(keyrelease) == "function" then keypress(0x31) task.wait(0.02) keyrelease(0x31)
            else
                pcall(function()
                    local vim = game:GetService("VirtualInputManager")
                    vim:SendKeyEvent(true, Enum.KeyCode.One, false, game) task.wait(0.02) vim:SendKeyEvent(false, Enum.KeyCode.One, false, game)
                end)
            end
        end)
    end

    local function setupAttackDetection()
        local function onToolEquipped(tool)
            if not tool or not tool:IsA("Tool") then return end
            local oldActivate = tool.Activate
            if oldActivate then
                local newActivate = function(...)
                    if getgenv().AntiParryEnabled then
                        isAttacking = true
                        attackEndTime = os.clock() + 0.3
                        task.delay(0.3, function()
                            if os.clock() >= attackEndTime then
                                isAttacking = false
                                stopStareAtTarget()
                            end
                        end)
                    end
                    return oldActivate(...)
                end
                local success = pcall(function() tool.Activate = newActivate end)
                if not success then
                    tool.Activated:Connect(function()
                        if getgenv().AntiParryEnabled then
                            isAttacking = true attackEndTime = os.clock() + 0.3
                            task.delay(0.3, function() if os.clock() >= attackEndTime then isAttacking = false end end)
                        end
                    end)
                end
            else
                tool.Activated:Connect(function()
                    if getgenv().AntiParryEnabled then
                        isAttacking = true attackEndTime = os.clock() + 0.3
                        task.delay(0.3, function()
                            if os.clock() >= attackEndTime then
                                isAttacking = false
                                stopStareAtTarget()
                            end
                        end)
                    end
                end)
            end
        end

        local function onCharacterAdded(char)
            local tool = char:FindFirstChildOfClass("Tool") if tool then onToolEquipped(tool) end
            local conn = char.ChildAdded:Connect(function(child) if child:IsA("Tool") then onToolEquipped(child) end end)
            regConn(conn)
        end

        if localPlayer.Character then onCharacterAdded(localPlayer.Character) end
        regConn(localPlayer.CharacterAdded:Connect(onCharacterAdded))
    end

    local function setupParryDetection()
        local parryKeywords = { "parry", "block", "deflect", "counter" }
        local function watchPlayer(player)
            if player == localPlayer then return end
            local function onChar(char)
                local hum = char:WaitForChild("Humanoid", 5) if not hum then return end
                local animator = hum:FindFirstChild("Animator") if not animator then return end
                local conn = animator.AnimationPlayed:Connect(function(track)
                    if not getgenv().AntiParryEnabled or not isAttacking then return end
                    local name, id = string.lower(track.Name or ""), string.lower(tostring(track.Animation and track.Animation.AnimationId or ""))
                    for _, keyword in pairs(parryKeywords) do
                        if string.find(name, keyword) or string.find(id, keyword) then unequipWeapon() isAttacking = false break end
                    end
                end)
                regConn(conn)

                local attrConn = char.AttributeAdded:Connect(function(attrName)
                    if not getgenv().AntiParryEnabled or not isAttacking then return end
                    if string.find(string.lower(attrName), "parry") and char:GetAttribute(attrName) == true then unequipWeapon() isAttacking = false end
                end)
                regConn(attrConn)
            end
            if player.Character then onChar(player.Character) end
            regConn(player.CharacterAdded:Connect(onChar))
        end
        for _, p in pairs(Players:GetPlayers()) do watchPlayer(p) end
        regConn(Players.PlayerAdded:Connect(watchPlayer))
    end

    local function setupManualAttackFlag()
        regConn(UserInputService.InputBegan:Connect(function(input, gameProcessed)
            if gameProcessed then return end
            if input.UserInputType == Enum.UserInputType.MouseButton1 and getgenv().AntiParryEnabled then
                isAttacking = true attackEndTime = os.clock() + 0.3
                task.delay(0.3, function()
                    if os.clock() >= attackEndTime then
                        isAttacking = false
                        stopStareAtTarget()
                    end
                end)
            end
        end))
    end

    setupAttackDetection() setupParryDetection() setupManualAttackFlag()
end

--[18. TARGET STRAFE (ОРБИТАЛЬНЫЙ АИМ)]--
local strafeAngle = 0
regConn(RunService.RenderStepped:Connect(function(dt)
    if not getgenv().TargetStrafeEnabled then return end
    local myChar = localPlayer.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHrp then return end

    local closestHrp = nil
    local minDist = math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 and not p.Character:GetAttribute("Downed") then
                local dist = (p.Character.HumanoidRootPart.Position - myHrp.Position).Magnitude
                if dist < minDist and dist <= 25 then
                    minDist = dist
                    closestHrp = p.Character.HumanoidRootPart
                end
            end
        end
    end

    if closestHrp then
        strafeAngle = (strafeAngle + dt * (getgenv().StrafeSpeed or 6)) % (math.pi * 2)
        local radius = getgenv().StrafeRadius or 8
        local offset = Vector3.new(math.cos(strafeAngle) * radius, 0, math.sin(strafeAngle) * radius)
        myHrp.CFrame = CFrame.new(closestHrp.Position + offset, closestHrp.Position)
    end
end))

        if getgenv().AutoExecuteEnabled then
            local myChar = localPlayer.Character
            local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if myHrp then
                for _, p in pairs(Players:GetPlayers()) do
                    if p ~= localPlayer and p.Character and p.Character:GetAttribute("Downed") == true then
                        local tHrp = p.Character:FindFirstChild("HumanoidRootPart")
                        if tHrp and (tHrp.Position - myHrp.Position).Magnitude <= 9 then
                            if type(keypress) == "function" then keypress(0x45) task.wait(0.02) keyrelease(0x45)
                            else
                                pcall(function()
                                    game:GetService("VirtualInputManager"):SendKeyEvent(true, Enum.KeyCode.E, false, game) task.wait(0.02)
                                    game:GetService("VirtualInputManager"):SendKeyEvent(false, Enum.KeyCode.E, false, game)
                                end)
                            end
                            break
                        end
                    end
                end
            end
        end
    end

local function applyWeaponTrail(tool)
    if not tool or not tool:IsA("Tool") or not getgenv().WeaponTrailEnabled then return end
    local handle = tool:FindFirstChild("Handle") or tool:FindFirstChildOfClass("Part") or tool:FindFirstChildOfClass("MeshPart")
    if not handle or handle:FindFirstChild("TrunkistTrail") then return end

    local a0 = Instance.new("Attachment", handle) a0.Position = Vector3.new(0, -1, 0)
    local a1 = Instance.new("Attachment", handle) a1.Position = Vector3.new(0, 2, 0)

    local trail = Instance.new("Trail")
    trail.Name = "TrunkistTrail"
    trail.Attachment0 = a0
    trail.Attachment1 = a1
    trail.Color = ColorSequence.new(Color3.fromRGB(190, 60, 255), Color3.fromRGB(255, 50, 150))
    trail.Lifetime = 0.35
    trail.WidthScale = NumberSequence.new({NumberSequenceKeypoint.new(0, 1), NumberSequenceKeypoint.new(1, 0)})
    trail.Parent = handle
end

regConn(localPlayer.CharacterAdded:Connect(function(char)
    char.ChildAdded:Connect(function(child)
        if child:IsA("Tool") then task.wait(0.1) applyWeaponTrail(child) end
    end)
end))

        pcall(function()

            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hrp and hum then
                local moveDir = hum.MoveDirection

                local vertical = 0
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then vertical = flySpeed end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then vertical = -flySpeed end
                hrp.Velocity = Vector3.new(velocity.X, vertical, velocity.Z)

                if char:GetAttribute("Downed") == true then char:SetAttribute("Downed", false) end

                    end

    end
